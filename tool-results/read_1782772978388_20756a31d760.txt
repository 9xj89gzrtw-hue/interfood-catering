   234→// ─── REVEAL ───
   235→function Reveal({ children, className = "", delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   236→  const ref = useRef<HTMLDivElement>(null);
   237→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   238→  return (
   239→    <motion.div ref={ref} initial="hidden" animate={isInView ? "visible" : "hidden"} variants={fadeUp} transition={{ delay }} className={className}>
   240→      {children}
   241→    </motion.div>
   242→  );
   243→}

   245→// ─── COUNTER ───
   246→function Counter({ target, suffix = "" }: { target: number; suffix?: string }) {
   247→  const [count, setCount] = useState(0);
   248→  const ref = useRef<HTMLSpanElement>(null);
   249→  const isInView = useInView(ref, { once: true });
   250→  useEffect(() => {
   251→    if (!isInView) return;
   252→    let start = 0;
   253→    const step = Math.max(Math.ceil(target / 60), 1);
   254→    const timer = setInterval(() => {
   255→      start += step;
   256→      if (start >= target) { setCount(target); clearInterval(timer); }
   257→      else setCount(start);
   258→    }, 30);
   259→    return () => clearInterval(timer);
   260→  }, [isInView, target]);
   261→  return <span ref={ref}>{count}{suffix}</span>;
   262→}
