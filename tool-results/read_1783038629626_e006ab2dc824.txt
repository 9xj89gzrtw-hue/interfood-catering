   527→export default function CinematicGallery() {
   528→  const isMobile = useIsMobile();
   529→  const sectionRef = useRef<HTMLElement>(null);
   530→  const headerInView = useInView(sectionRef, { once: true, margin: "-80px" });
   531→  const prefersReduced = useReducedMotion();
   532→
   533→  const [activeIndex, setActiveIndex] = useState(0);
   534→  const [isHovering, setIsHovering] = useState(false);
   535→  const [isTouching, setIsTouching] = useState(false);
   536→  const [progress, setProgress] = useState(0);
   537→  const [lightboxIndex, setLightboxIndex] = useState<number | null>(null);
   538→
   539→  const isAutoplaying = !isHovering && !isTouching && !prefersReduced && lightboxIndex === null;
   540→
   541→  /* Drag state */
   542→  const dragStartX = useRef(0);
   543→  const isDragging = useRef(false);
   544→  const hasDragged = useRef(false);
   545→  const dragVelocity = useRef(0);
   546→  const lastDragX = useRef(0);
   547→  const lastDragTime = useRef(0);
   548→
   549→  /* Autoplay */
   550→  useEffect(() => {
   551→    if (!isAutoplaying) return;
   552→
   553→    const interval = setInterval(() => {
   554→      setProgress(0);
   555→      setActiveIndex((prev) => (prev + 1) % GALLERY.length);
   556→    }, AUTOPLAY_INTERVAL);
   557→
   558→    return () => clearInterval(interval);
   559→  }, [isAutoplaying]);
   560→
   561→  /* Progress bar animation — uses rAF callback for setState (not in effect body) */

   563→    if (!isAutoplaying) return;

   565→    let start: number | null = null;
   566→    let rafId: number;
   567→
   568→    const animate = (timestamp: number) => {
   569→      if (!start) start = timestamp;
   570→      const elapsed = timestamp - start;
   571→      const p = Math.min(elapsed / AUTOPLAY_INTERVAL, 1);
   572→      setProgress(p);
   573→
   574→      if (p < 1) {
   575→        rafId = requestAnimationFrame(animate);
   576→      }
   577→    };
   578→
   579→    rafId = requestAnimationFrame(animate);
   580→    return () => cancelAnimationFrame(rafId);

   603→      hasDragged.current = false;
   604→      dragVelocity.current = 0;
   605→      (e.target as HTMLElement).setPointerCapture?.(e.pointerId);
   606→    },
   607→    []
   608→  );
   609→
   610→  const handlePointerMove = useCallback(
   611→    (e: React.PointerEvent<HTMLDivElement>) => {
   612→      if (!isDragging.current) return;
   613→      const dx = e.clientX - dragStartX.current;
   614→      const now = Date.now();
   615→      const dt = now - lastDragTime.current;
   616→
   617→      /* Track velocity for momentum */
   618→      if (dt > 0) {
   619→        dragVelocity.current = (e.clientX - lastDragX.current) / dt * 1000;
   620→      }
   621→      lastDragX.current = e.clientX;
   622→      lastDragTime.current = now;
   623→
   624→      if (Math.abs(dx) > 10) hasDragged.current = true;
   625→    },
   626→    []
   627→  );
   628→
   629→  const handlePointerUp = useCallback(
   630→    (e: React.PointerEvent<HTMLDivElement>) => {
   631→      if (!isDragging.current) return;
   632→      isDragging.current = false;
   633→
   634→      const dx = e.clientX - dragStartX.current;
   635→      const velocity = dragVelocity.current;
   636→
   637→      if (hasDragged.current) {
   638→        /* Use momentum: if velocity is high, go next/prev even with small drag */
   639→        if (dx > 50 || velocity > 300) goPrev();
   640→        else if (dx < -50 || velocity < -300) goNext();
   641→      }
   642→
   643→      dragVelocity.current = 0;
   644→    },
   645→    [goNext, goPrev]
   646→  );
   647→

   650→    setIsTouching(true);
   651→  }, []);

   661→
   662→  const lightboxNext = useCallback(() => {

   665→
   666→  const lightboxClose = useCallback(() => {

   668→  }, []);
   669→

   672→  }, []);
   673→
   674→  /* Mobile swipe handler for stack cards */

   677→    else goPrev();
   678→  }, [goNext, goPrev]);
   679→

   709→            width: `${((activeIndex + (isAutoplaying ? progress : 0)) / GALLERY.length) * 100}%`,
   710→            transition: "width 0.1s linear",
   711→          }}
   712→        />
   713→      </div>
   714→
   715→      {/* Gold accent line above gallery */}
   716→      <div
   717→        style={{
   718→          width: "min(80px, 15vw)",
   719→          height: 1,
   720→          background:
   721→            "linear-gradient(90deg, transparent, rgba(184,134,11,0.4), transparent)",
   722→          margin: "0 auto 2rem",
   723→        }}
   724→      />
   725→
   726→      <div
   727→        className="container"
   728→        style={{ position: "relative", zIndex: 1 }}
   729→      >
   730→        {/* ── Section header ── */}
   731→        <motion.div
   732→          initial={{ opacity: 0, y: 30 }}
   733→          animate={headerInView ? { opacity: 1, y: 0 } : {}}
   734→          transition={{ duration: 0.8, ease: EASE_PREMIUM }}
   735→          style={{ marginBottom: "2.5rem", textAlign: "center" }}
   736→        >
   737→          <div
   738→            style={{
   739→              display: "flex",
   740→              alignItems: "center",
   741→              justifyContent: "center",
   742→              gap: "0.75rem",
   743→              marginBottom: "1.25rem",

   746→            <span
   747→              style={{
   748→                width: 32,
   749→                height: 1,
   750→                background:
   751→                  "linear-gradient(90deg, transparent, rgba(184,134,11,0.3))",

   803→
   804→        {/* ── Coverflow carousel (Desktop) / Stack (Mobile) ── */}
   805→        <motion.div

   941→                        activeIndex === i
   942→                          ? "linear-gradient(90deg, #B8860B, #D4A63E)"
   943→                          : "rgba(184,134,11,0.2)",
   944→                      border: "none",

   961→                aria-label="Next image"
   962→                style={{
   963→                  width: 44,
