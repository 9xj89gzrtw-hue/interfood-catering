   198→   198→  function validate(): FormErrors {

   201→   201→    if (!form.phone.trim()) {
   202→   202→      errs.phone = "Введите номер телефона";
   203→   203→    } else {
   204→   204→      const phoneClean = form.phone.replace(/[\s\-\(\)]/g, "");
   205→   205→      if (!/^(\+7|7|8)\d{10}$/.test(phoneClean)) {
   206→   206→        errs.phone = "Формат: +7 (XXX) XXX-XX-XX";
   207→   207→      }
   208→   208→    }
   209→   209→    if (form.guests) {
   210→   210→      const n = parseInt(form.guests, 10);
   211→   211→      if (isNaN(n) || n < 1) errs.guests = "Введите количество персон";
   212→   212→    }
   213→   213→    return errs;

   217→   217→  function validateField(name: keyof FormErrors) {

   232→   232→          delete newErrors.phone;

   234→   234→      }

   236→   236→    if (name === "guests" && form.guests) {

   247→   247→  /* ─── Phone number masking ─── */
   248→   248→  function formatPhone(value: string): string {
   249→   249→    const digits = value.replace(/\D/g, "");
   250→   250→    if (!digits) return "";
   251→   251→    let result = "";
   252→   252→    if (digits.startsWith("8")) {
   253→   253→      result = "+7";
   254→   254→    } else if (digits.startsWith("7")) {
   255→   255→      result = "+7";
   256→   256→    } else if (digits.startsWith("9") || digits.startsWith("3")) {
   257→   257→      result = "+7" + digits[0];
   258→   258→    } else {
   259→   259→      return value; // let them type whatever
   260→   260→    }
   261→   261→
   262→   262→    const rest = digits.startsWith("8") || digits.startsWith("7")
   263→   263→      ? digits.slice(1)
   264→   264→      : digits;
   265→   265→
   266→   266→    if (rest.length > 0) result += " (" + rest.slice(0, 3);
   267→   267→    if (rest.length >= 3) result += ") ";
   268→   268→    if (rest.length > 3) result += rest.slice(3, 6);
   269→   269→    if (rest.length > 6) result += "-" + rest.slice(6, 8);
   270→   270→    if (rest.length > 8) result += "-" + rest.slice(8, 10);
   271→   271→
   272→   272→    return result;
   273→   273→  }

   318→   318→      setSubmitted(true);
   319→   319→      setForm(INITIAL_FORM);
   320→   320→      setErrors({});

   323→   323→    } finally {

   325→   325→    }
