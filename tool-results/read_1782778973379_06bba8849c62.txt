   262→// ─── REVEAL ───
   263→function Reveal({ children, className = "", delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   264→  const ref = useRef<HTMLDivElement>(null);
   265→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   266→  return (
   267→    <motion.div ref={ref} initial="hidden" animate={isInView ? "visible" : "hidden"} variants={fadeUp} transition={{ delay }} className={className}>
   268→      {children}
   269→    </motion.div>
   270→  );
   271→}

   273→// ─── COUNTER ───
   274→function Counter({ target, suffix = "" }: { target: number; suffix?: string }) {
   275→  const [count, setCount] = useState(0);
   276→  const ref = useRef<HTMLSpanElement>(null);
   277→  const isInView = useInView(ref, { once: true });
   278→  useEffect(() => {
   279→    if (!isInView) return;
   280→    let start = 0;
   281→    const step = Math.max(Math.ceil(target / 60), 1);
   282→    const timer = setInterval(() => {
   283→      start += step;
   284→      if (start >= target) { setCount(target); clearInterval(timer); }
   285→      else setCount(start);
   286→    }, 30);
   287→    return () => clearInterval(timer);
   288→  }, [isInView, target]);
   289→  return <span ref={ref}>{count}{suffix}</span>;
   290→}
