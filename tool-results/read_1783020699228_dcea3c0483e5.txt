    72→function MorphingText({ words, interval = 2800 }: { words: string[]; interval?: number }) {
    73→  const [index, setIndex] = useState(0);
    74→  const [visible, setVisible] = useState(true);
    75→
    76→  useEffect(() => {
    77→    const timer = setInterval(() => {
    78→      setVisible(false);
    79→      setTimeout(() => {
    80→        setIndex((prev) => (prev + 1) % words.length);
    81→        setVisible(true);
    82→      }, 450);
    83→    }, interval);
    84→    return () => clearInterval(timer);
    85→  }, [words.length, interval]);
    86→
    87→  return (
    88→    <AnimatePresence mode="wait">
    89→      <motion.span
    90→        key={words[index]}
    91→        initial={{ opacity: 0, filter: "blur(12px)", y: 8, scale: 0.92 }}
    92→        animate={
    93→          visible
    94→            ? { opacity: 1, filter: "blur(0px)", y: 0, scale: 1 }
    95→            : { opacity: 0, filter: "blur(12px)", y: -8, scale: 1.08 }
    96→        }
    97→        exit={{ opacity: 0, filter: "blur(12px)", y: -8, scale: 1.08 }}
    98→        transition={{ duration: 0.5, ease: EASE_PREMIUM }}
    99→        style={{
   100→          display: "inline-block",
   101→          fontStyle: "italic",
   102→          color: "#D4A63E",
   103→          // Fixed min-width based on longest word to prevent layout shift
   104→          minWidth: `${LONGEST_TAGLINE.length + 0.5}ch`,
   105→          textAlign: "center" as const,
   106→          willChange: "opacity, filter, transform",
   107→          textShadow:
   108→            "0 2px 12px rgba(0,0,0,0.6), 0 0 40px rgba(184,134,11,0.3), 0 0 80px rgba(184,134,11,0.15)",
   109→        }}
   110→      >
   111→        {words[index]}
   112→      </motion.span>
   113→    </AnimatePresence>
   114→  );
   115→}

   630→// ─── Floating Particles — Enhanced for mobile visibility ────
   631→function FloatingParticles() {
   632→  // Generate particles via useMemo since component is only rendered client-side (gated by `mounted`)
   633→  const particles = useMemo(() => {
   634→    const isMob = typeof window !== "undefined" && window.innerWidth < 768;
   635→    const count = isMob ? 12 : 16;
   636→    return Array.from({ length: count }, (_, i) => ({
   637→      id: i,
   638→      x: Math.random() * 100,
   639→      size: isMob ? 2 + Math.random() * 3.5 : 2 + Math.random() * 3,
   640→      duration: 7 + Math.random() * 12,
   641→      delay: Math.random() * 8,
   642→      opacity: isMob ? 0.35 + Math.random() * 0.4 : 0.2 + Math.random() * 0.4,
   643→      glow: isMob ? 8 + Math.random() * 10 : 4 + Math.random() * 8,
   644→    }));
   645→  }, []);
   646→
   647→  if (particles.length === 0) return null;
   648→
   649→  return (
   650→    <>
   651→      {particles.map((p) => (
   652→        <div
   653→          key={p.id}
   654→          style={{
   655→            position: "absolute",
   656→            left: `${p.x}%`,
   657→            bottom: "-5%",
   658→            width: p.size,
   659→            height: p.size,
   660→            borderRadius: "50%",
   661→            background: "rgba(184,134,11,0.9)",

   663→            opacity: 0,

   669→      ))}
   670→    </>
   671→  );

   674→// ─── Mouse-Following Glow (desktop only) ──────────────────
   675→function MouseGlow() {
   676→  const glowX = useMotionValue(0);
   677→  const glowY = useMotionValue(0);
   678→  const springX = useSpring(glowX, { stiffness: 80, damping: 30 });
   679→  const springY = useSpring(glowY, { stiffness: 80, damping: 30 });
   680→  const [isActive, setIsActive] = useState(false);
   681→
   682→  const glowBackground = useTransform(
   683→    [springX, springY],
   684→    ([x, y]: number[]) =>
   685→      `radial-gradient(600px circle at ${x}px ${y}px, rgba(184,134,11,0.08), transparent 60%)`
   686→  );
   687→
   688→  useEffect(() => {
   689→    if (typeof window !== "undefined" && "ontouchstart" in window) return;
   690→    const handleMouseMove = (e: MouseEvent) => {
   691→      glowX.set(e.clientX);
   692→      glowY.set(e.clientY);
   693→      setIsActive(true);
   694→    };
   695→    const handleMouseLeave = () => setIsActive(false);
   696→    window.addEventListener("mousemove", handleMouseMove);
   697→    document.addEventListener("mouseleave", handleMouseLeave);
   698→    return () => {
   699→      window.removeEventListener("mousemove", handleMouseMove);
   700→      document.removeEventListener("mouseleave", handleMouseLeave);
   701→    };
   702→  }, [glowX, glowY]);
   703→
   704→  if (!isActive) return null;
   705→
   706→  return (
   707→    <motion.div
   708→      style={{
   709→        position: "absolute",
   710→        inset: 0,
   711→        pointerEvents: "none",
   712→        zIndex: 2,
   713→        background: glowBackground,
   714→        transition: "opacity 0.5s",
   715→      }}
   716→    />
   717→  );
   718→}
