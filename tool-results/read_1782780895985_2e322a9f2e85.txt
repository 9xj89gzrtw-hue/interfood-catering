   129→function Reveal({ children, className = "", delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   130→  const ref = useRef<HTMLDivElement>(null);
   131→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   132→  return (
   133→    <motion.div
   134→      ref={ref}
   135→      initial="hidden"
   136→      animate={isInView ? "visible" : "hidden"}
   137→      variants={fadeUp}
   138→      transition={{ delay }}
   139→      className={className}
   140→    >
   141→      {children}
   142→    </motion.div>
   143→  );
   144→}
