    84→function Reveal({ children, delay = 0, className = "" }: { children: React.ReactNode; delay?: number; className?: string }) {
    85→  const ref = useRef<HTMLDivElement>(null);
    86→  const inView = useInView(ref, { once: true, margin: "-60px" });
    87→  return (
    88→    <motion.div
    89→      ref={ref}
    90→      className={className}
    91→      initial={{ opacity: 0, y: 40 }}
    92→      animate={inView ? { opacity: 1, y: 0 } : {}}
    93→      transition={{ duration: 0.8, delay, ease: [0.25, 1, 0.5, 1] as const }}
    94→    >
    95→      {children}
    96→    </motion.div>
    97→  );
    98→}

   426→    setLightboxIndex(index);
   427→    setLightbox({ src: filteredImages[index].src, alt: filteredImages[index].alt, desc: filteredImages[index].desc });
   428→  };
   429→
   430→  const closeLightbox = useCallback(() => {
   431→    setLightbox(null);
   432→  }, []);
   433→
   434→  const navigateLightbox = useCallback((direction: "prev" | "next") => {
   435→    setLightboxIndex((prevIndex) => {
   436→      const newIndex = direction === "next"
   437→        ? (prevIndex + 1) % filteredImages.length
   438→        : (prevIndex - 1 + filteredImages.length) % filteredImages.length;
   439→      setLightbox({ src: filteredImages[newIndex].src, alt: filteredImages[newIndex].alt, desc: filteredImages[newIndex].desc });
   440→      return newIndex;
   441→    });
   442→  }, [filteredImages]);

   444→  // Keyboard navigation for lightbox
   445→  useEffect(() => {
   446→    const handleKeyDown = (e: KeyboardEvent) => {
   447→      if (!lightbox) return;
   448→      if (e.key === "Escape") closeLightbox();
   449→      if (e.key === "ArrowRight") navigateLightbox("next");
   450→      if (e.key === "ArrowLeft") navigateLightbox("prev");
   451→    };
   452→    window.addEventListener("keydown", handleKeyDown);
   453→    return () => window.removeEventListener("keydown", handleKeyDown);
   454→  }, [lightbox, closeLightbox, navigateLightbox]);
   455→
   456→  // Lock body scroll when lightbox is open
   457→  useEffect(() => {
   458→    document.body.style.overflow = lightbox ? "hidden" : "";
   459→    return () => { document.body.style.overflow = ""; };
   460→  }, [lightbox]);
   461→
   462→  // Hero parallax
   463→  const heroRef = useRef<HTMLDivElement>(null);
   464→  const { scrollYProgress: heroScroll } = useScroll({
   465→    target: heroRef,
   466→    offset: ["start start", "end start"],
   467→  });
   468→  const heroY = useTransform(heroScroll, [0, 1], ["0%", "30%"]);
   469→  const heroOpacity = useTransform(heroScroll, [0, 0.8], [1, 0]);
   470→
   471→  return (
   472→    <main>
   473→      <SiteNav />
   474→
   475→      {/* ═══ 1. VIDEO HERO ═══ */}
   476→      <section ref={heroRef} className="hero" aria-label="Галерея проектов" style={{ minHeight: "85vh" }}>
   477→        <motion.div style={{ position: "absolute", inset: 0, y: heroY }}>
   478→          <video
   479→            autoPlay
   480→            muted
   481→            loop
   482→            playsInline
   483→            preload="metadata"
   484→            poster="/images/poster_hero.jpg"
   485→            aria-label="Видео-фон: кейтеринг Интерфуд"
   486→            style={{ width: "100%", height: "100%", objectFit: "cover" }}
   487→          >

   489→          </video>
   490→        </motion.div>
   491→        <div
   492→          style={{
   493→            position: "absolute", inset: 0, zIndex: 1,
   494→            background: "linear-gradient(to bottom, rgba(254,253,251,0.1) 0%, rgba(254,253,251,0.05) 30%, rgba(254,253,251,0.25) 60%, rgba(254,253,251,0.92) 100%)",
   495→          }}
