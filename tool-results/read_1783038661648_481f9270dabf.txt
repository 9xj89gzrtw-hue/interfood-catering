   316→/* ═══════════════════════════════════════════════════════════════════════════
   317→   ANIMATED COUNTER — Counts up/down to a target value
   318→   ═══════════════════════════════════════════════════════════════════════════ */
   319→
   320→function AnimatedCounter({ value, prefix = "", suffix = "" }: { value: number; prefix?: string; suffix?: string }) {
   321→  const spring = useSpring(0, { stiffness: 120, damping: 25 });
   322→  const display = useTransform(spring, (v) => {
   323→    const rounded = Math.round(v);
   324→    return prefix + rounded.toLocaleString("ru-RU") + suffix;
   325→  });
   326→
   327→  useEffect(() => {
   328→    spring.set(value);
   329→  }, [value, spring]);
   330→
   331→  return <motion.span>{display}</motion.span>;
   332→}

   354→
   355→function DishCard({
   356→  dish,
   357→  inCart,
   358→  onAdd,
   359→  onRemove,
   360→  onUpdateQty,
   361→  activeCategory,
   362→  isMobile,
   363→  index,
   364→}: {
   365→  dish: DishItem;
   366→  inCart: number;
   367→  onAdd: (dish: DishItem, categoryId: string) => void;
   368→  onRemove: (dishId: string) => void;
   369→  onUpdateQty: (dishId: string, delta: number) => void;
   370→  activeCategory: string;
   371→  isMobile: boolean;
   372→  index: number;
   373→}) {
   374→  const [imgLoaded, setImgLoaded] = useState(false);
   375→  const [isHovered, setIsHovered] = useState(false);
   376→
   377→  // 3D tilt motion values
   378→  const x = useMotionValue(0);
   379→  const y = useMotionValue(0);
   380→  const rotateX = useTransform(y, [-0.5, 0.5], [6, -6]);
   381→  const rotateY = useTransform(x, [-0.5, 0.5], [-6, 6]);
   382→
   383→  /** Handle mouse move for 3D tilt effect */
   384→  const handleMouseMove = useCallback(
   385→    (e: React.MouseEvent<HTMLDivElement>) => {
   386→      if (isMobile) return;
   387→      const rect = e.currentTarget.getBoundingClientRect();
   388→      const px = (e.clientX - rect.left) / rect.width - 0.5;
   389→      const py = (e.clientY - rect.top) / rect.height - 0.5;
   390→      x.set(px);
   391→      y.set(py);
   392→    },
   393→    [isMobile, x, y]
   394→  );
   395→
   396→  const handleMouseLeave = useCallback(() => {
   397→    x.set(0);
   398→    y.set(0);
   399→    setIsHovered(false);
   400→  }, [x, y]);
   401→
   402→  return (
   403→    <motion.div
   404→      initial={{ opacity: 0, y: 30, scale: 0.97 }}
   405→      animate={{ opacity: 1, y: 0, scale: 1 }}
   406→      transition={{
   407→        duration: 0.5,
   408→        delay: index * 0.08,
   409→        ease: [0.25, 0.46, 0.45, 0.94],
   410→      }}
   411→      className="group relative"
   412→      style={{
   413→        perspective: "800px",
   414→      }}
   415→    >
   416→      <motion.div
   417→        style={{
   418→          rotateX: isMobile ? 0 : rotateX,
   419→          rotateY: isMobile ? 0 : rotateY,
   420→          transformStyle: "preserve-3d",
   421→        }}
   422→        onMouseMove={handleMouseMove}
   423→        onMouseEnter={() => setIsHovered(true)}
   424→        onMouseLeave={handleMouseLeave}
   425→        className="relative overflow-hidden"
   426→        /* ─── Card shell ─── */
   427→      >
   428→        {/* Animated gold gradient border on hover */}
   429→        <motion.div
   430→          animate={{
   431→            opacity: isHovered || inCart > 0 ? 1 : 0,
   432→          }}
   433→          transition={{ duration: 0.4 }}
   434→          className="absolute inset-0 rounded-2xl"
   435→          style={{
   436→            padding: "2px",
   437→            background: isHovered
   438→              ? "linear-gradient(135deg, #B8860B, #D4A63E, #E5BF65, #D4A63E, #B8860B)"
   439→              : inCart > 0
   440→                ? "#B8860B"
   441→                : "transparent",
   442→            backgroundSize: isHovered ? "300% 300%" : "100% 100%",
   443→            animation: isHovered ? "goldBorderShift 3s ease infinite" : "none",
   444→          }}
   445→        >
   446→          <div
   447→            className="w-full h-full rounded-2xl"
   448→            style={{ background: COLORS.card }}
   449→          />
   450→        </motion.div>
   451→
   452→        {/* Inner card content */}
   453→        <div
   454→          className="relative rounded-2xl overflow-hidden"
   455→          style={{
   456→            background: COLORS.card,
   457→            boxShadow: isHovered
   458→              ? "0 20px 60px rgba(0,0,0,0.12), 0 8px 24px rgba(184,134,11,0.08)"
   459→              : "0 2px 12px rgba(0,0,0,0.04)",
   460→            transition: "box-shadow 0.4s ease",
   461→          }}
   462→        >
   463→          <div
   464→            className={`flex ${isMobile ? "flex-col" : "flex-col sm:flex-row"}`}
   465→          >
   466→            {/* ─── Image section ─── */}
   467→            <div
   468→              className={`relative overflow-hidden shrink-0 ${
   469→                isMobile ? "h-52" : "h-52 sm:h-auto sm:w-48 lg:w-56"
   470→              }`}
   471→              style={{ transformStyle: "preserve-3d" }}
   472→            >
   473→              {!imgLoaded && <ImageSkeleton />}
   474→              <Image
   475→                src={dish.image}
   476→                alt={dish.name}
   477→                fill
   478→                className={`object-cover transition-transform duration-700 group-hover:scale-110 ${
   479→                  imgLoaded ? "opacity-100" : "opacity-0"

   499→                    style={{
