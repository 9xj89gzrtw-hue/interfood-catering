    70→function ParticleCanvas({ isMobile }: { isMobile: boolean }) {
    71→  const canvasRef = useRef<HTMLCanvasElement>(null);
    72→  const animRef = useRef<number>(0);
    73→
    74→  useEffect(() => {
    75→    const canvas = canvasRef.current;
    76→    if (!canvas) return;
    77→    const ctx = canvas.getContext("2d");
    78→    if (!ctx) return;
    79→
    80→    const count = isMobile ? 18 : 35;
    81→    const size = isMobile ? 3 : 2;
    82→    interface Particle {
    83→      x: number;
    84→      y: number;
    85→      r: number;
    86→      vx: number;
    87→      vy: number;
    88→      opacity: number;
    89→      phase: number;
    90→    }
    91→    const particles: Particle[] = [];
    92→
    93→    const resize = () => {
    94→      canvas.width = canvas.offsetWidth * devicePixelRatio;
    95→      canvas.height = canvas.offsetHeight * devicePixelRatio;
    96→      ctx.scale(devicePixelRatio, devicePixelRatio);
    97→    };
    98→    resize();
    99→    window.addEventListener("resize", resize);
   100→
   101→    const w = canvas.offsetWidth;
   102→    const h = canvas.offsetHeight;
   103→    for (let i = 0; i < count; i++) {
   104→      particles.push({
   105→        x: Math.random() * w,
   106→        y: Math.random() * h,
   107→        r: Math.random() * size + 1,
   108→        vx: (Math.random() - 0.5) * 0.3,
   109→        vy: (Math.random() - 0.5) * 0.2 - 0.1,
   110→        opacity: 0.15 + Math.random() * 0.1,
   111→        phase: Math.random() * Math.PI * 2,
   112→      });
   113→    }
   114→
   115→    const draw = (time: number) => {
   116→      ctx.clearRect(0, 0, canvas.offsetWidth, canvas.offsetHeight);
   117→      for (const p of particles) {
   118→        p.x += p.vx;
   119→        p.y += p.vy;
   120→        const pulse = Math.sin(time * 0.001 + p.phase) * 0.05;
   121→        const alpha = Math.max(0.05, p.opacity + pulse);
   122→
   123→        if (p.x < -10) p.x = canvas.offsetWidth + 10;
   124→        if (p.x > canvas.offsetWidth + 10) p.x = -10;
   125→        if (p.y < -10) p.y = canvas.offsetHeight + 10;
   126→        if (p.y > canvas.offsetHeight + 10) p.y = -10;
   127→
   128→        ctx.beginPath();
   129→        ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
   130→        ctx.fillStyle = `rgba(212, 166, 62, ${alpha})`;
   131→        ctx.fill();
   132→      }
   133→      animRef.current = requestAnimationFrame(draw);
   134→    };
   135→    animRef.current = requestAnimationFrame(draw);
   136→
   137→    return () => {
   138→      window.removeEventListener("resize", resize);
   139→      cancelAnimationFrame(animRef.current);
   140→    };
   141→  }, [isMobile]);
   142→
   143→  return (
   144→    <canvas
   145→      ref={canvasRef}
   146→      aria-hidden="true"
   147→      style={{
   148→        position: "absolute",
   149→        inset: 0,
   150→        width: "100%",
   151→        height: "100%",
   152→        pointerEvents: "none",
   153→        zIndex: 3,
   154→      }}
   155→    />
   156→  );
   157→}

   162→function GlitchText({ text, active }: { text: string; active: boolean }) {
   163→  const [displayed, setDisplayed] = useState(text);
   164→  const [glitching, setGlitching] = useState(false);
   165→  const prevText = useRef(text);
   166→
   167→  useEffect(() => {
   168→    if (prevText.current === text) return;
   169→    prevText.current = text;
   170→    setGlitching(true);
   171→
   172→    const totalSteps = 8;
   173→    let step = 0;
   174→    const interval = setInterval(() => {
   175→      step++;
   176→      if (step >= totalSteps) {
   177→        setDisplayed(text);
   178→        setGlitching(false);
   179→        clearInterval(interval);
   180→        return;
   181→      }
   182→      // Scramble: progressively more characters become correct
   183→      const progress = step / totalSteps;
   184→      const scrambled = text
   185→        .split("")
   186→        .map((char, i) => {
   187→          if (char === " ") return " ";
   188→          // Characters before the progress threshold are correct
   189→          if (i / text.length < progress * 0.8) return char;
   190→          // Random scramble character
   191→          return SCRAMBLE_CHARS[Math.floor(Math.random() * SCRAMBLE_CHARS.length)];
   192→        })
   193→        .join("");
   194→      setDisplayed(scrambled);
   195→    }, 40);
   196→
   197→    return () => clearInterval(interval);
   198→  }, [text]);
   199→
   200→  return (
   201→    <span
   202→      style={{
   203→        display: "inline-block",
   204→        fontStyle: "italic",
   205→        position: "relative",
   206→        color: glitching ? "#E5BF65" : "#D4A63E",
   207→        textShadow: glitching
   208→          ? "0 0 8px rgba(212,166,62,0.6), -1px 0 rgba(229,191,101,0.4), 1px 0 rgba(184,134,11,0.4)"
   209→          : "none",
   210→        transition: "color 0.2s, text-shadow 0.2s",
   211→      }}
   212→    >
   213→      {displayed}
   214→      {glitching && (
   215→        <>
   216→          <span
   217→            aria-hidden="true"
   218→            style={{
   219→              position: "absolute",
   220→              left: -2,
   221→              top: 0,
   222→              color: "rgba(184,134,11,0.5)",
   223→              clipPath: "inset(20% 0 60% 0)",
   224→            }}
   225→          >
   226→            {displayed}
   227→          </span>
   228→          <span
   229→            aria-hidden="true"
   230→            style={{
   231→              position: "absolute",
   232→              left: 2,
   233→              top: 0,
   234→              color: "rgba(229,191,101,0.5)",
   235→              clipPath: "inset(60% 0 10% 0)",
   236→            }}
   237→          >
   238→            {displayed}
   239→          </span>
   240→        </>
   241→      )}
   242→    </span>
   243→  );
   244→}

   298→// ═══════════════════════════════════════════════════════════════
   299→//  Magnetic CTA Button — Enhanced with stretchy + particle burst
   300→// ═══════════════════════════════════════════════════════════════
   301→function MagneticButton({
   302→  children,
   303→  href,
   304→  variant = "primary",
   305→  isMobile,
   306→  prefersReduced,
   307→}: {
   308→  children: React.ReactNode;
   309→  href: string;
   310→  variant?: "primary" | "secondary";
   311→  isMobile: boolean;
   312→  prefersReduced: boolean;
   313→}) {
   314→  const ref = useRef<HTMLAnchorElement>(null);
   315→  const [ripples, setRipples] = useState<{ x: number; y: number; id: number }[]>([]);
   316→  const x = useMotionValue(0);
   317→  const y = useMotionValue(0);
   318→  const springX = useSpring(x, SPRING_MAGNETIC);
   319→  const springY = useSpring(y, SPRING_MAGNETIC);
   320→  const [hovered, setHovered] = useState(false);
   321→
   322→  // Stretchy effect: track cursor offset for scaleX distortion
   323→  const stretchX = useMotionValue(1);
   324→  const stretchY = useMotionValue(1);
   325→  const springStretchX = useSpring(stretchX, { stiffness: 200, damping: 20 });
   326→  const springStretchY = useSpring(stretchY, { stiffness: 200, damping: 20 });
   327→
   328→  const handleMouseMove = useCallback(
   329→    (e: React.MouseEvent) => {
   330→      if (isMobile || prefersReduced) return;
   331→      const el = ref.current;
   332→      if (!el) return;
   333→      const rect = el.getBoundingClientRect();
   334→      const cx = rect.left + rect.width / 2;
   335→      const cy = rect.top + rect.height / 2;
   336→      const dist = Math.hypot(e.clientX - cx, e.clientY - cy);
   337→      const radius = 140; // stronger magnetic radius
   338→      if (dist < radius) {
   339→        const strength = 1 - dist / radius;
   340→        // Stronger pull factor
   341→        x.set((e.clientX - cx) * strength * 0.45);
   342→        y.set((e.clientY - cy) * strength * 0.45);
   343→        // Stretchy: elongate toward cursor direction
   344→        const dx = (e.clientX - cx) / rect.width;
   345→        const dy = (e.clientY - cy) / rect.height;
   346→        stretchX.set(1 + Math.abs(dx) * strength * 0.08);
   347→        stretchY.set(1 + Math.abs(dy) * strength * 0.06);
   348→      } else {
   349→        x.set(0);
   350→        y.set(0);
   351→        stretchX.set(1);
   352→        stretchY.set(1);
   353→      }
   354→    },
   355→    [isMobile, prefersReduced, x, y, stretchX, stretchY]
   356→  );
   357→
   358→  const handleMouseLeave = useCallback(() => {
   359→    x.set(0);
   360→    y.set(0);
   361→    stretchX.set(1);
   362→    stretchY.set(1);
   363→    setHovered(false);
   364→  }, [x, y, stretchX, stretchY]);
   365→
   366→  const handleClick = useCallback(
   367→    (e: React.MouseEvent) => {
   368→      const el = ref.current;
   369→      if (!el) return;
   370→      const rect = el.getBoundingClientRect();
   371→      const id = Date.now();
   372→      setRipples((prev) => [...prev, { x: e.clientX - rect.left, y: e.clientY - rect.top, id }]);
   373→      setTimeout(() => setRipples((prev) => prev.filter((r) => r.id !== id)), 600);
   374→    },
   375→    []
   376→  );
   377→
   378→  const isPrimary = variant === "primary";
   379→
   380→  return (
   381→    <motion.a
   382→      ref={ref}
   383→      href={href}
   384→      onMouseMove={handleMouseMove}
   385→      onMouseLeave={handleMouseLeave}
   386→      onMouseEnter={() => setHovered(true)}
   387→      onClick={handleClick}
   388→      style={{
   389→        x: springX,
   390→        y: springY,
   391→        scaleX: springStretchX,
   392→        scaleY: springStretchY,
   393→        position: "relative",
   394→        overflow: "visible", // allow particles to escape
   395→        display: "inline-flex",
   396→        alignItems: "center",
   397→        justifyContent: "center",
   398→        gap: "0.5rem",
   399→        minWidth: isMobile ? "100%" : "auto",
   400→        padding: isMobile ? "1rem 1.5rem" : "1rem 2.25rem",
   401→        minHeight: 48,
   402→        borderRadius: "0.5rem",
   403→        fontFamily: "var(--font-sans)",
   404→        fontWeight: 500,
   405→        fontSize: "0.95rem",
   406→        letterSpacing: "0.02em",
   407→        textDecoration: "none",
   408→        cursor: "pointer",
   409→        border: "none",
   410→        outline: "none",
   411→        background: isPrimary
   412→          ? hovered
   413→            ? "linear-gradient(135deg, #D4A63E, #B8860B)"
   414→            : "linear-gradient(135deg, #B8860B, #9A6F0A)"
   415→          : "transparent",
   416→        color: isPrimary ? "#FFFFFF" : "#B8860B",
   417→        boxShadow: isPrimary
   418→          ? hovered
   419→            ? "0 0 30px rgba(184,134,11,0.35), 0 4px 15px rgba(184,134,11,0.25)"
   420→            : "0 4px 15px rgba(184,134,11,0.2)"
   421→          : hovered
   422→            ? "0 0 20px rgba(184,134,11,0.15)"
   423→            : "none",
   424→        borderWidth: isPrimary ? 0 : 2,
   425→        borderStyle: "solid",
   426→        borderColor: "#B8860B",
   427→        willChange: "transform",
   428→        transition: "box-shadow 0.3s ease, background 0.3s ease",
   429→      }}
   430→      whileHover={prefersReduced ? {} : { scale: 1.02 }}
   431→      whileTap={{ scale: 0.98 }}
   432→    >
   433→      {/* Clip overflow for button content but not particles */}
   434→      <span
   435→        style={{
   436→          display: "inline-flex",
   437→          alignItems: "center",
   438→          justifyContent: "center",
   439→          gap: "0.5rem",
   440→          overflow: "hidden",
   441→          borderRadius: "0.5rem",
   442→          position: "relative",
   443→        }}
   444→      >
   445→        {ripples.map((ripple) => (
   446→          <motion.span
   447→            key={ripple.id}
   448→            initial={{ scale: 0, opacity: 0.4 }}
   449→            animate={{ scale: 4, opacity: 0 }}
   450→            transition={{ duration: 0.6, ease: "easeOut" }}
   451→            style={{
   452→              position: "absolute",
   453→              left: ripple.x,
   454→              top: ripple.y,
   455→              width: 20,
   456→              height: 20,
   457→              marginLeft: -10,
   458→              marginTop: -10,
   459→              borderRadius: "50%",
   460→              background: isPrimary ? "rgba(255,255,255,0.5)" : "rgba(184,134,11,0.3)",
   461→              pointerEvents: "none",
   462→            }}
   463→          />
   464→        ))}
   465→        {children}
   466→      </span>
   467→      {/* Gold Particle Burst on hover (primary only) */}
   468→      {isPrimary && !isMobile && <GoldParticleBurst active={hovered && !prefersReduced} />}
   469→    </motion.a>
   470→  );
   471→}

   473→// ═══════════════════════════════════════════════════════════════
   474→//  Cursor-Following Light Spot (flashlight effect)
   475→// ═══════════════════════════════════════════════════════════════
   476→function CursorLightSpot({ containerRef }: { containerRef: React.RefObject<HTMLElement | null> }) {
   477→  const lightRef = useRef<HTMLDivElement>(null);
   478→
   479→  useEffect(() => {
   480→    const container = containerRef.current;
   481→    if (!container) return;
   482→
   483→    const handleMouseMove = (e: MouseEvent) => {
   484→      const rect = container.getBoundingClientRect();
   485→      const x = e.clientX - rect.left;
   486→      const y = e.clientY - rect.top;
   487→      if (lightRef.current) {
   488→        lightRef.current.style.transform = `translate(${x}px, ${y}px)`;
   489→        lightRef.current.style.opacity = "1";
   490→      }
   491→    };
   492→
   493→    const handleMouseLeave = () => {
   494→      if (lightRef.current) {
   495→        lightRef.current.style.opacity = "0";
   496→      }
   497→    };
   498→
   499→    container.addEventListener("mousemove", handleMouseMove);
   500→    container.addEventListener("mouseleave", handleMouseLeave);
   501→    return () => {
   502→      container.removeEventListener("mousemove", handleMouseMove);
   503→      container.removeEventListener("mouseleave", handleMouseLeave);
   504→    };
   505→  }, [containerRef]);
   506→
   507→  return (
