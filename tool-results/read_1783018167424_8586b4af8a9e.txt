   179→function Reveal({
   180→  children,
   181→  className = "",
   182→  delay = 0,
   183→}: {
   184→  children: React.ReactNode;
   185→  className?: string;
   186→  delay?: number;
   187→}) {
   188→  const ref = useRef<HTMLDivElement>(null);
   189→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   190→  return (
   191→    <motion.div
   192→      ref={ref}
   193→      initial="hidden"
   194→      animate={isInView ? "visible" : "hidden"}
   195→      variants={fadeUp}
   196→      transition={{ delay }}
   197→      className={className}
   198→    >
   199→      {children}
   200→    </motion.div>
   201→  );
   202→}
