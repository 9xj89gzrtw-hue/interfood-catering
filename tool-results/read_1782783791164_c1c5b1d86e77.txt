   123→function Reveal({
   124→  children,
   125→  className = "",
   126→  delay = 0,
   127→}: {
   128→  children: React.ReactNode;
   129→  className?: string;
   130→  delay?: number;
   131→}) {
   132→  const ref = useRef<HTMLDivElement>(null);
   133→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   134→  return (
   135→    <motion.div
   136→      ref={ref}
   137→      initial="hidden"
   138→      animate={isInView ? "visible" : "hidden"}
   139→      variants={fadeUp}
   140→      transition={{ delay }}
   141→      className={className}
   142→    >
   143→      {children}
   144→    </motion.div>
   145→  );
   146→}

   192→  /* ─── Validation ─── */
   193→  function validate(): FormErrors {
   194→    const errs: FormErrors = {};
   195→    if (!form.name.trim()) errs.name = "Введите ваше имя";
   196→    if (!form.phone.trim()) {
   197→      errs.phone = "Введите номер телефона";
   198→    } else {
   199→      const phoneClean = form.phone.replace(/[\s\-\(\)]/g, "");
   200→      if (!/^(\+7|7|8)\d{10}$/.test(phoneClean)) {
   201→        errs.phone = "Введите корректный номер телефона";
   202→      }
   203→    }
   204→    if (form.email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) {
   205→      errs.email = "Введите корректный email";
   206→    }
   207→    if (form.guests) {
   208→      const n = parseInt(form.guests, 10);
   209→      if (isNaN(n) || n < 1) errs.guests = "Введите количество гостей";
   210→    }
   211→    return errs;
   212→  }
