   295→/* ─── Reveal on Scroll Helper ─── */
   296→function Reveal({
   297→  children,
   298→  className = "",
   299→  delay = 0,
   300→}: {
   301→  children: React.ReactNode;
   302→  className?: string;
   303→  delay?: number;
   304→}) {
   305→  const ref = useRef<HTMLDivElement>(null);
   306→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   307→  return (
   308→    <motion.div
   309→      ref={ref}
   310→      initial="hidden"
   311→      animate={isInView ? "visible" : "hidden"}
   312→      variants={fadeUp}
   313→      transition={{ delay }}
   314→      className={className}
   315→    >
   316→      {children}
   317→    </motion.div>
   318→  );
   319→}
