   151→/* ─── Animated Price Hook ─── */
   152→
   153→function useAnimatedPrice(target: number) {
   154→  const motionVal = useMotionValue(0);
   155→  const spring = useSpring(motionVal, { stiffness: 120, damping: 30 });
   156→  const display = useTransform(spring, (v: number) => Math.round(v));
   157→  const [shown, setShown] = useState("0");
   158→
   159→  useEffect(() => {
   160→    motionVal.set(target);
   161→  }, [target, motionVal]);
   162→
   163→  useEffect(() => {
   164→    const unsub = display.on("change", (v: number) => {
   165→      setShown(v.toLocaleString("ru-RU"));
   166→    });
   167→    return unsub;
   168→  }, [display]);
   169→
   170→  return shown;
   171→}
   172→
   173→/* ─── Animated Count Hook (for guest count etc.) ─── */
   174→
   175→function useAnimatedCount(target: number) {
   176→  const motionVal = useMotionValue(0);
   177→  const spring = useSpring(motionVal, { stiffness: 300, damping: 35 });
   178→  const display = useTransform(spring, (v: number) => Math.round(v));
   179→  const [shown, setShown] = useState(String(target));
   180→
   181→  useEffect(() => {
   182→    motionVal.set(target);
   183→  }, [target, motionVal]);
   184→
   185→  useEffect(() => {
   186→    const unsub = display.on("change", (v: number) => {
   187→      setShown(String(v));
   188→    });
   189→    return unsub;
   190→  }, [display]);
   191→
   192→  return shown;
   193→}

   195→/* ─── Reveal Component ─── */
   196→
   197→const fadeUp = {
   198→  hidden: { opacity: 0, y: 60 },
   199→  visible: { opacity: 1, y: 0, transition: { duration: 0.8, ease: [0.4, 0, 0.2, 1] as const } },
   200→};
   201→
   202→function Reveal({
   203→  children,
   204→  className = "",
   205→  delay = 0,
   206→}: {
   207→  children: React.ReactNode;
   208→  className?: string;
   209→  delay?: number;
   210→}) {
   211→  const ref = useRef<HTMLDivElement>(null);
   212→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   213→  return (
   214→    <motion.div
   215→      ref={ref}
   216→      initial="hidden"
   217→      animate={isInView ? "visible" : "hidden"}
   218→      variants={fadeUp}
   219→      transition={{ delay }}
   220→      className={className}
   221→    >
   222→      {children}
   223→    </motion.div>
   224→  );
   225→}
