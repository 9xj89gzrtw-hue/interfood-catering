    81→gate() {
    82→  local stage="$1" name="$2" result="$3" critical="${4:-no}" what="${5:-}" how="${6:-}" tool="${7:-}" evidence="${8:-}"
    83→  TOTAL=$((TOTAL + 1))
    84→
    85→  local icon status_str
    86→  if [[ "$result" == "PASS" ]]; then
    87→    icon="✅"; PASSED=$((PASSED + 1)); status_str="PASS"
    88→    STAGES_PASSED="$STAGES_PASSED $stage"
    89→  elif [[ "$result" == "FAIL" ]]; then
    90→    icon="❌"; FAILED=$((FAILED + 1)); status_str="FAIL"
    91→    STAGES_FAILED="$STAGES_FAILED $stage"
    92→    if [[ "$critical" == "yes" ]]; then
    93→      CRITICAL_FAIL=$((CRITICAL_FAIL + 1))
    94→      status_str="FAIL-CRITICAL"
    95→    fi
    96→  elif [[ "$result" == "SKIP" ]]; then
    97→    icon="⏭️"; SKIPPED=$((SKIPPED + 1)); status_str="SKIP"
    98→    STAGES_SKIPPED="$STAGES_SKIPPED $stage"
    99→  else
   100→    icon="⚠️"; status_str="UNKNOWN"
   101→  fi
   102→
   103→  printf "  %s [%s] %s  (crit=%s)\n" "$icon" "$stage" "$name" "$critical"
   104→  if [[ -n "$evidence" ]]; then
   105→    printf "      Evidence: %s\n" "$evidence"
   106→  fi
   107→
   108→  # Запись в детали отчёта
   109→  DETAILS="${DETAILS}## ${stage}: ${name} — ${status_str}\n\n"
   110→  DETAILS="${DETAILS}| Поле | Значение |\n|------|----------|\n"
   111→  DETAILS="${DETAILS}| Результат | ${icon} ${status_str} |\n"
   112→  DETAILS="${DETAILS}| Critical | ${critical} |\n"
   113→  DETAILS="${DETAILS}| ЧТО проверяет | ${what} |\n"
   114→  DETAILS="${DETAILS}| КАК проверяет | ${how} |\n"
   115→  DETAILS="${DETAILS}| ЧЕМ проверяет | ${tool} |\n"
   116→  if [[ -n "$evidence" ]]; then
   117→    DETAILS="${DETAILS}| Evidence | ${evidence} |\n"
   118→  fi
   119→  DETAILS="${DETAILS}\n"
   120→}
