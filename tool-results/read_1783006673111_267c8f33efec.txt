    13→/* ─── Анимация появления секции при скролле ─── */
    14→function RevealSection({
    15→  children,
    16→  delay = 0,
    17→  className = "",
    18→}: {
    19→  children: React.ReactNode;
    20→  delay?: number;
    21→  className?: string;
    22→}) {
    23→  const ref = useRef<HTMLDivElement>(null);
    24→  const inView = useInView(ref, { once: true, margin: "-80px" });
    25→
    26→  return (
    27→    <motion.div
    28→      ref={ref}
    29→      initial={{ opacity: 0, y: 40 }}
    30→      animate={inView ? { opacity: 1, y: 0 } : { opacity: 0, y: 40 }}
    31→      transition={{ duration: 0.7, delay, ease: [0.25, 1, 0.5, 1] as const }}
    32→      className={className}
    33→    >
    34→      {children}
    35→    </motion.div>
    36→  );
    37→}
