   152→   152→/* ─── Animated Price Hook ─── */
   153→   153→
   154→   154→function useAnimatedPrice(target: number) {
   155→   155→  const motionVal = useMotionValue(0);
   156→   156→  const spring = useSpring(motionVal, { stiffness: 120, damping: 30 });
   157→   157→  const display = useTransform(spring, (v: number) => Math.round(v));
   158→   158→  const [shown, setShown] = useState("0");
   159→   159→
   160→   160→  useEffect(() => {
   161→   161→    motionVal.set(target);
   162→   162→  }, [target, motionVal]);
   163→   163→
   164→   164→  useEffect(() => {
   165→   165→    const unsub = display.on("change", (v: number) => {
   166→   166→      setShown(v.toLocaleString("ru-RU"));
   167→   167→    });
   168→   168→    return unsub;
   169→   169→  }, [display]);
   170→   170→
   171→   171→  return shown;
   172→   172→}
   173→   173→
   174→   174→/* ─── Animated Count Hook (for guest count etc.) ─── */
   175→   175→
   176→   176→function useAnimatedCount(target: number) {
   177→   177→  const motionVal = useMotionValue(0);
   178→   178→  const spring = useSpring(motionVal, { stiffness: 300, damping: 35 });
   179→   179→  const display = useTransform(spring, (v: number) => Math.round(v));
   180→   180→  const [shown, setShown] = useState(String(target));
   181→   181→
   182→   182→  useEffect(() => {
   183→   183→    motionVal.set(target);
   184→   184→  }, [target, motionVal]);
   185→   185→
   186→   186→  useEffect(() => {
   187→   187→    const unsub = display.on("change", (v: number) => {
   188→   188→      setShown(String(v));
   189→   189→    });
   190→   190→    return unsub;
   191→   191→  }, [display]);
   192→   192→
   193→   193→  return shown;
   194→   194→}

   203→   203→function Reveal({
   204→   204→  children,
   205→   205→  className = "",
   206→   206→  delay = 0,
   207→   207→}: {
   208→   208→  children: React.ReactNode;
   209→   209→  className?: string;
   210→   210→  delay?: number;
   211→   211→}) {
   212→   212→  const ref = useRef<HTMLDivElement>(null);
   213→   213→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   214→   214→  return (
   215→   215→    <motion.div
   216→   216→      ref={ref}
   217→   217→      initial="hidden"
   218→   218→      animate={isInView ? "visible" : "hidden"}
   219→   219→      variants={fadeUp}
   220→   220→      transition={{ delay }}
   221→   221→      className={className}
   222→   222→    >
   223→   223→      {children}
   224→   224→    </motion.div>
   225→   225→  );
   226→   226→}
