   165→// ─── Reveal wrapper (uses framer-motion as progressive enhancement) ───
   166→function Reveal({ children, delay = 0, className = "" }: { children: React.ReactNode; delay?: number; className?: string }) {
   167→  const ref = useRef<HTMLDivElement>(null);
   168→  const inView = useInView(ref, { once: true, margin: "-60px" });
   169→  return (
   170→    <motion.div
   171→      ref={ref}
   172→      className={className}
   173→      initial={{ opacity: 0, y: 50 }}
   174→      animate={inView ? { opacity: 1, y: 0 } : {}}
   175→      transition={{ duration: 0.8, delay, ease: [0.25, 1, 0.5, 1] }}
   176→    >
   177→      {children}
   178→    </motion.div>
   179→  );
   180→}
