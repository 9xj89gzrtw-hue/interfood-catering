   166→   166→function Reveal({ children, className = "", delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   167→   167→  const ref = useRef<HTMLDivElement>(null);
   168→   168→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   169→   169→  return (
   170→   170→    <motion.div
   171→   171→      ref={ref}
   172→   172→      initial="hidden"
   173→   173→      animate={isInView ? "visible" : "hidden"}
   174→   174→      variants={fadeUp}
   175→   175→      transition={{ delay }}
   176→   176→      className={className}
   177→   177→    >
   178→   178→      {children}
   179→   179→    </motion.div>
   180→   180→  );
   181→   181→}
