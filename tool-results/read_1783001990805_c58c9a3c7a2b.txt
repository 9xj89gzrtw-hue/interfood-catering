   246→   246→/* ─── Reveal on Scroll Helper ─── */
   247→   247→function Reveal({
   248→   248→  children,
   249→   249→  className = "",
   250→   250→  delay = 0,
   251→   251→}: {
   252→   252→  children: React.ReactNode;
   253→   253→  className?: string;
   254→   254→  delay?: number;
   255→   255→}) {
   256→   256→  const ref = useRef<HTMLDivElement>(null);
   257→   257→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   258→   258→  return (
   259→   259→    <motion.div
   260→   260→      ref={ref}
   261→   261→      initial="hidden"
   262→   262→      animate={isInView ? "visible" : "hidden"}
   263→   263→      variants={fadeUp}
   264→   264→      transition={{ delay }}
   265→   265→      className={className}
   266→   266→    >
   267→   267→      {children}
   268→   268→    </motion.div>
   269→   269→  );
   270→   270→}
