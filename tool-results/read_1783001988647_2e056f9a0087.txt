   246→/* ─── Reveal on Scroll Helper ─── */
   247→function Reveal({
   248→  children,
   249→  className = "",
   250→  delay = 0,
   251→}: {
   252→  children: React.ReactNode;
   253→  className?: string;
   254→  delay?: number;
   255→}) {
   256→  const ref = useRef<HTMLDivElement>(null);
   257→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   258→  return (
   259→    <motion.div
   260→      ref={ref}
   261→      initial="hidden"
   262→      animate={isInView ? "visible" : "hidden"}
   263→      variants={fadeUp}
   264→      transition={{ delay }}
   265→      className={className}
   266→    >
   267→      {children}
   268→    </motion.div>
   269→  );
   270→}
