   162→// ─── REVEAL ANIMATION WRAPPER ───
   163→function Reveal({
   164→  children,
   165→  delay = 0,
   166→  direction = "up",
   167→}: {
   168→  children: React.ReactNode;
   169→  delay?: number;
   170→  direction?: "up" | "left" | "right";
   171→}) {
   172→  const ref = useRef<HTMLDivElement>(null);
   173→  const inView = useInView(ref, { once: true, margin: "-60px" });
   174→  const offsets = { up: { y: 40 }, left: { x: -40 }, right: { x: 40 } };
   175→
   176→  return (
   177→    <motion.div
   178→      ref={ref}
   179→      initial={{ opacity: 0, ...offsets[direction] }}
   180→      animate={inView ? { opacity: 1, x: 0, y: 0 } : {}}
   181→      transition={{
   182→        duration: 0.8,
   183→        delay,
   184→        ease: [0.25, 1, 0.5, 1] as const,
   185→      }}
   186→    >
   187→      {children}
   188→    </motion.div>
   189→  );
   190→}

   364→}: {

   367→}) {
   368→  const [rotation, setRotation] = useState(0);
   369→  const [zoom, setZoom] = useState(1);
   370→  const [isDragging, setIsDragging] = useState(false);
   371→  const lastX = useRef(0);
   372→
   373→  const handlePointerDown = useCallback(
   374→    (e: React.MouseEvent | React.TouchEvent) => {
   375→      setIsDragging(true);
   376→      const clientX = "touches" in e ? e.touches[0].clientX : e.clientX;
   377→      lastX.current = clientX;
   378→    },
   379→    []
   380→  );
   381→
   382→  const handlePointerMove = useCallback(
   383→    (e: React.MouseEvent | React.TouchEvent) => {
   384→      if (!isDragging) return;
   385→      const clientX = "touches" in e ? e.touches[0].clientX : e.clientX;
   386→      const delta = clientX - lastX.current;
   387→      setRotation((prev) => prev + delta * 0.4);
   388→      lastX.current = clientX;
   389→    },
   390→    [isDragging]
   391→  );
   392→
   393→  const handlePointerUp = useCallback(() => {
   394→    setIsDragging(false);
   395→  }, []);
   396→
   397→  return (
   398→    <motion.div
   399→      initial={{ opacity: 0 }}
   400→      animate={{ opacity: 1 }}

   535→function TourModal({ venue, onClose }: { venue: Venue | null; onClose: () => void }) {
   536→  useEffect(() => {
   537→    if (venue) {
   538→      document.body.style.overflow = "hidden";
   539→    } else {
   540→      document.body.style.overflow = "";
   541→    }
   542→    return () => { document.body.style.overflow = ""; };
   543→  }, [venue]);
   544→
   545→  useEffect(() => {
   546→    const onKey = (e: KeyboardEvent) => { if (e.key === "Escape") onClose(); };
   547→    window.addEventListener("keydown", onKey);
   548→    return () => window.removeEventListener("keydown", onKey);
   549→  }, [onClose]);
   550→
   551→  return (
   552→    <AnimatePresence>
   553→      {venue && <TourModalInner key={venue.id} venue={venue} onClose={onClose} />}
   554→    </AnimatePresence>
   555→  );
   556→}
