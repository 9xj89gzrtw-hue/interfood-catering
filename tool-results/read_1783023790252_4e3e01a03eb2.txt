    66→function ParticleCanvas({ isMobile }: { isMobile: boolean }) {
    67→  const canvasRef = useRef<HTMLCanvasElement>(null);
    68→  const animRef = useRef<number>(0);
    69→
    70→  useEffect(() => {
    71→    const canvas = canvasRef.current;
    72→    if (!canvas) return;
    73→    const ctx = canvas.getContext("2d");
    74→    if (!ctx) return;
    75→
    76→    const count = isMobile ? 18 : 35;
    77→    const size = isMobile ? 3 : 2;
    78→    interface Particle {
    79→      x: number;
    80→      y: number;
    81→      r: number;
    82→      vx: number;
    83→      vy: number;
    84→      opacity: number;
    85→      phase: number;
    86→    }
    87→    const particles: Particle[] = [];
    88→
    89→    const resize = () => {
    90→      canvas.width = canvas.offsetWidth * devicePixelRatio;
    91→      canvas.height = canvas.offsetHeight * devicePixelRatio;
    92→      ctx.scale(devicePixelRatio, devicePixelRatio);
    93→    };
    94→    resize();
    95→    window.addEventListener("resize", resize);
    96→
    97→    // Initialize particles
    98→    const w = canvas.offsetWidth;
    99→    const h = canvas.offsetHeight;
   100→    for (let i = 0; i < count; i++) {
   101→      particles.push({
   102→        x: Math.random() * w,
   103→        y: Math.random() * h,
   104→        r: Math.random() * size + 1,
   105→        vx: (Math.random() - 0.5) * 0.3,
   106→        vy: (Math.random() - 0.5) * 0.2 - 0.1,
   107→        opacity: 0.15 + Math.random() * 0.1,
   108→        phase: Math.random() * Math.PI * 2,
   109→      });
   110→    }
   111→
   112→    const draw = (time: number) => {
   113→      ctx.clearRect(0, 0, canvas.offsetWidth, canvas.offsetHeight);
   114→      for (const p of particles) {
   115→        p.x += p.vx;
   116→        p.y += p.vy;
   117→        const pulse = Math.sin(time * 0.001 + p.phase) * 0.05;
   118→        const alpha = Math.max(0.05, p.opacity + pulse);
   119→
   120→        // Wrap around
   121→        if (p.x < -10) p.x = canvas.offsetWidth + 10;
   122→        if (p.x > canvas.offsetWidth + 10) p.x = -10;
   123→        if (p.y < -10) p.y = canvas.offsetHeight + 10;
   124→        if (p.y > canvas.offsetHeight + 10) p.y = -10;
   125→
   126→        ctx.beginPath();
   127→        ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
   128→        ctx.fillStyle = `rgba(212, 166, 62, ${alpha})`;
   129→        ctx.fill();
   130→      }
   131→      animRef.current = requestAnimationFrame(draw);
   132→    };
   133→    animRef.current = requestAnimationFrame(draw);
   134→
   135→    return () => {
   136→      window.removeEventListener("resize", resize);
   137→      cancelAnimationFrame(animRef.current);
   138→    };
   139→  }, [isMobile]);
   140→
   141→  return (
   142→    <canvas
   143→      ref={canvasRef}
   144→      aria-hidden="true"
   145→      style={{
   146→        position: "absolute",
   147→        inset: 0,
   148→        width: "100%",
   149→        height: "100%",
   150→        pointerEvents: "none",
   151→        zIndex: 3,
   152→      }}
   153→    />
   154→  );
   155→}

   160→function MagneticButton({
   161→  children,
   162→  href,
   163→  variant = "primary",
   164→  isMobile,
   165→}: {
   166→  children: React.ReactNode;
   167→  href: string;
   168→  variant?: "primary" | "secondary";
   169→  isMobile: boolean;
   170→}) {
   171→  const ref = useRef<HTMLAnchorElement>(null);
   172→  const [ripples, setRipples] = useState<{ x: number; y: number; id: number }[]>([]);
   173→  const x = useMotionValue(0);
   174→  const y = useMotionValue(0);
   175→  const springX = useSpring(x, SPRING_MAGNETIC);
   176→  const springY = useSpring(y, SPRING_MAGNETIC);
   177→  const [hovered, setHovered] = useState(false);
   178→
   179→  const handleMouseMove = useCallback(
   180→    (e: React.MouseEvent) => {
   181→      if (isMobile) return;
   182→      const el = ref.current;
   183→      if (!el) return;
   184→      const rect = el.getBoundingClientRect();
   185→      const cx = rect.left + rect.width / 2;
   186→      const cy = rect.top + rect.height / 2;
   187→      const dist = Math.hypot(e.clientX - cx, e.clientY - cy);
   188→      const radius = 120;
   189→      if (dist < radius) {
   190→        const strength = 1 - dist / radius;
   191→        x.set((e.clientX - cx) * strength * 0.3);
   192→        y.set((e.clientY - cy) * strength * 0.3);
   193→      } else {
   194→        x.set(0);
   195→        y.set(0);
   196→      }
   197→    },
   198→    [isMobile, x, y]
   199→  );
   200→
   201→  const handleMouseLeave = useCallback(() => {
   202→    x.set(0);
   203→    y.set(0);
   204→    setHovered(false);
   205→  }, [x, y]);
   206→
   207→  const handleClick = useCallback(
   208→    (e: React.MouseEvent) => {
   209→      const el = ref.current;
   210→      if (!el) return;
   211→      const rect = el.getBoundingClientRect();
   212→      const id = Date.now();
   213→      setRipples((prev) => [...prev, { x: e.clientX - rect.left, y: e.clientY - rect.top, id }]);
   214→      setTimeout(() => setRipples((prev) => prev.filter((r) => r.id !== id)), 600);
   215→    },
   216→    []
   217→  );
   218→
   219→  const isPrimary = variant === "primary";
   220→
   221→  return (
   222→    <motion.a
   223→      ref={ref}
   224→      href={href}
   225→      onMouseMove={handleMouseMove}
   226→      onMouseLeave={handleMouseLeave}
   227→      onMouseEnter={() => setHovered(true)}
   228→      onClick={handleClick}
   229→      style={{
   230→        x: springX,
   231→        y: springY,
   232→        position: "relative",
   233→        overflow: "hidden",
   234→        display: "inline-flex",
   235→        alignItems: "center",
   236→        justifyContent: "center",
   237→        gap: "0.5rem",
   238→        minWidth: isMobile ? "100%" : "auto",
   239→        padding: isMobile ? "1rem 1.5rem" : "1rem 2.25rem",
   240→        minHeight: 48,
   241→        borderRadius: "0.5rem",
   242→        fontFamily: "var(--font-sans)",
   243→        fontWeight: 500,
   244→        fontSize: "0.95rem",
   245→        letterSpacing: "0.02em",
   246→        textDecoration: "none",
   247→        cursor: "pointer",
   248→        border: "none",
   249→        outline: "none",
   250→        background: isPrimary
   251→          ? hovered
   252→            ? "linear-gradient(135deg, #D4A63E, #B8860B)"
   253→            : "linear-gradient(135deg, #B8860B, #9A6F0A)"
   254→          : "transparent",
   255→        color: isPrimary ? "#FFFFFF" : "#B8860B",
   256→        boxShadow: isPrimary
   257→          ? hovered
   258→            ? "0 0 30px rgba(184,134,11,0.35), 0 4px 15px rgba(184,134,11,0.25)"
   259→            : "0 4px 15px rgba(184,134,11,0.2)"
   260→          : hovered
   261→            ? "0 0 20px rgba(184,134,11,0.15)"
   262→            : "none",
   263→        borderWidth: isPrimary ? 0 : 2,
   264→        borderStyle: "solid",
   265→        borderColor: "#B8860B",
   266→        willChange: "transform",
   267→        transition: "box-shadow 0.3s ease, background 0.3s ease",
   268→      }}
   269→      whileHover={{ scale: 1.02 }}
   270→      whileTap={{ scale: 0.98 }}
   271→    >
   272→      {ripples.map((ripple) => (
   273→        <motion.span
   274→          key={ripple.id}
   275→          initial={{ scale: 0, opacity: 0.4 }}
   276→          animate={{ scale: 4, opacity: 0 }}
   277→          transition={{ duration: 0.6, ease: "easeOut" }}
   278→          style={{
   279→            position: "absolute",
   280→            left: ripple.x,
   281→            top: ripple.y,
   282→            width: 20,
   283→            height: 20,
   284→            marginLeft: -10,
   285→            marginTop: -10,
   286→            borderRadius: "50%",
   287→            background: isPrimary ? "rgba(255,255,255,0.5)" : "rgba(184,134,11,0.3)",
   288→            pointerEvents: "none",
   289→          }}
   290→        />
   291→      ))}
   292→      {children}
   293→    </motion.a>
   294→  );
   295→}
