   170→// ─── Spring & easing presets ────────────────────────────────────
   171→const EASE_PREMIUM = [0.16, 1, 0.3, 1] as const;
   172→const SPRING_SNAP = { type: "spring" as const, stiffness: 300, damping: 28 };

   176→// ═══════════════════════════════════════════════════════════════
   177→//  Magnetic Link — subtle cursor-follow on desktop
   178→// ═══════════════════════════════════════════════════════════════
   179→function MagneticLink({
   180→  children,
   181→  href,
   182→  style,
   183→  className,
   184→  onClick,
   185→  isMobile,
   186→  isActive,
   187→}: {
   188→  children: React.ReactNode;
   189→  href: string;
   190→  style?: React.CSSProperties;
   191→  className?: string;
   192→  onClick?: (e: React.MouseEvent<HTMLAnchorElement>) => void;
   193→  isMobile: boolean;
   194→  isActive?: boolean;
   195→}) {
   196→  const ref = useRef<HTMLAnchorElement>(null);
   197→  const x = useMotionValue(0);
   198→  const y = useMotionValue(0);
   199→  const springX = useSpring(x, SPRING_MAGNETIC_NAV);
   200→  const springY = useSpring(y, SPRING_MAGNETIC_NAV);
   201→
   202→  const handleMouseMove = useCallback(
   203→    (e: React.MouseEvent) => {
   204→      if (isMobile) return;
   205→      const el = ref.current;
   206→      if (!el) return;
   207→      const rect = el.getBoundingClientRect();
   208→      const cx = rect.left + rect.width / 2;
   209→      const cy = rect.top + rect.height / 2;
   210→      const dist = Math.hypot(e.clientX - cx, e.clientY - cy);
   211→      const radius = 80;
   212→      if (dist < radius) {
   213→        const strength = 1 - dist / radius;
   214→        x.set((e.clientX - cx) * strength * 0.15);
   215→        y.set((e.clientY - cy) * strength * 0.1);
   216→      } else {
   217→        x.set(0);
   218→        y.set(0);
   219→      }
   220→    },
   221→    [isMobile, x, y]
   222→  );
   223→
   224→  const handleMouseLeave = useCallback(() => {
   225→    x.set(0);
   226→    y.set(0);
   227→  }, [x, y]);
   228→
   229→  return (
   230→    <motion.a
   231→      ref={ref}
   232→      href={href}
   233→      onClick={onClick}
   234→      onMouseMove={handleMouseMove}
   235→      onMouseLeave={handleMouseLeave}
   236→      className={`${className || ""} ${isActive ? "nav-link-active" : ""}`}
   237→      style={{
   238→        ...style,
   239→        x: springX,
   240→        y: springY,
   241→        display: "inline-block",
   242→        willChange: isMobile ? "auto" : "transform",
   243→        fontSize: isActive ? "0.82rem" : style?.fontSize,
   244→        transition: style?.transition
   245→          ? `${style.transition}, font-size 0.3s cubic-bezier(0.16, 1, 0.3, 1)`
   246→          : "font-size 0.3s cubic-bezier(0.16, 1, 0.3, 1)",
   247→      }}
   248→    >
   249→      {children}
   250→      {/* Active section gold underline (persistent) */}
   251→      {isActive && (
   252→        <motion.span
   253→          layoutId="nav-active-underline"
   254→          style={{
   255→            position: "absolute",
   256→            bottom: 0,
   257→            left: 0,
   258→            right: 0,
   259→            height: 2,
   260→            background: "#B8860B",
   261→            borderRadius: 1,
   262→          }}
   263→          transition={{ type: "spring", stiffness: 300, damping: 25 }}
   264→        />
   265→      )}
   266→    </motion.a>
   267→  );
   268→}

   365→export default function SiteNav() {
   366→  const [scrolled, setScrolled] = useState(false);
   367→  const [menuOpen, setMenuOpen] = useState(false);
   368→  const [dropdownOpen, setDropdownOpen] = useState(false);
   369→  const dropdownRef = useRef<HTMLDivElement>(null);
   370→  const burgerRef = useRef<HTMLButtonElement>(null);
   371→  const navRef = useRef<HTMLElement>(null);
   372→  const isMobile = useIsMobile();
   373→
   374→  // ─── Active section tracking via IntersectionObserver ───
   375→  const [activeSection, setActiveSection] = useState<string | null>(null);
   376→
   377→  useEffect(() => {
   378→    const observer = new IntersectionObserver(
   379→      (entries) => {
   380→        // Find the most visible section
   381→        const visible = entries
   382→          .filter((e) => e.isIntersecting)
   383→          .sort((a, b) => b.intersectionRatio - a.intersectionRatio);
   384→        if (visible.length > 0) {
   385→          setActiveSection(visible[0].target.id);
   386→        }
   387→      },
   388→      { threshold: [0.2, 0.5], rootMargin: "-80px 0px -30% 0px" }
   389→    );
   390→
   391→    // Observe sections after a short delay (wait for page to render)
   392→    const timer = setTimeout(() => {
   393→      SECTION_IDS.forEach((id) => {
   394→        const el = document.getElementById(id);
   395→        if (el) observer.observe(el);
   396→      });
   397→    }, 500);
   398→
   399→    return () => {
   400→      clearTimeout(timer);
   401→      observer.disconnect();
   402→    };
   403→  }, []);
   404→
   405→  // ─── Scroll progress ───
   406→  const { scrollYProgress } = useScroll();
   407→  const scrollProgressWidth = useTransform(scrollYProgress, [0, 1], ["0%", "100%"]);
   408→
   409→  // Derived color values
   410→  const textColor = scrolled ? "#1A1714" : "#FFFFFF";
   411→  const textShadow = scrolled ? "none" : "2px 4px 12px rgba(0,0,0,0.7), 0 1px 3px rgba(0,0,0,0.4)";
   412→
   413→  // ─── Scroll listener ──────────────────────────────────────────
   414→  useEffect(() => {
   415→    const onScroll = () => setScrolled(window.scrollY > 60);
   416→    onScroll();
   417→    window.addEventListener("scroll", onScroll, { passive: true });
   418→    return () => window.removeEventListener("scroll", onScroll);
   419→  }, []);
   420→
   421→  // ─── Body overflow when menu open ────────────────────────────
   422→  useEffect(() => {
   423→    if (menuOpen) {
   424→      document.body.style.overflow = "hidden";
   425→    } else {
   426→      document.body.style.overflow = "";
   427→    }
   428→    return () => {
   429→      document.body.style.overflow = "";
   430→    };
   431→  }, [menuOpen]);
   432→
   433→  // ─── Close handlers ──────────────────────────────────────────
   434→  const closeMenu = useCallback(() => setMenuOpen(false), []);
   435→  const closeDropdown = useCallback(() => setDropdownOpen(false), []);
   436→
   437→  // ─── Escape key closes both ──────────────────────────────────
   438→  useEffect(() => {
   439→    const handleKeyDown = (e: KeyboardEvent) => {
   440→      if (e.key === "Escape") {
   441→        if (menuOpen) closeMenu();
   442→        if (dropdownOpen) closeDropdown();
   443→      }
   444→    };
   445→    document.addEventListener("keydown", handleKeyDown);
   446→    return () => document.removeEventListener("keydown", handleKeyDown);
   447→  }, [menuOpen, dropdownOpen, closeMenu, closeDropdown]);
   448→
   449→  // ─── Click outside closes dropdown ───────────────────────────
   450→  useEffect(() => {
   451→    if (!dropdownOpen) return;
   452→    const handleClick = (e: MouseEvent) => {
   453→      if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node)) {
   454→        closeDropdown();
   455→      }
   456→    };
   457→    document.addEventListener("mousedown", handleClick);
   458→    return () => document.removeEventListener("mousedown", handleClick);
   459→  }, [dropdownOpen, closeDropdown]);
   460→
   461→  // ─── Burger line color ───────────────────────────────────────
   462→  const burgerLineColor = menuOpen ? "#1A1714" : scrolled ? "#1A1714" : "#FFFFFF";
   463→
   464→  // ─── Helper: check if link matches active section ───
   465→  const isLinkActive = useCallback(
   466→    (href: string) => {
   467→      if (href.startsWith("#")) {
   468→        return activeSection === href.slice(1);
   469→      }
   470→      return false;
   471→    },
   472→    [activeSection]
   473→  );
   474→
   475→  return (
   476→    <>
   477→      {/* ═══════════════════════════════════════════════════════════
   478→          NAV BAR — fixed, z-9999, transparent on hero → glass on scroll
   479→          ═══════════════════════════════════════════════════════════ */}
   480→      <nav
   481→        ref={navRef}
   482→        role="navigation"
