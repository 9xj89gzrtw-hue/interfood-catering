   258→// ─── REVEAL ───
   259→function Reveal({ children, className = "", delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   260→  const ref = useRef<HTMLDivElement>(null);
   261→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   262→  return (
   263→    <motion.div ref={ref} initial="hidden" animate={isInView ? "visible" : "hidden"} variants={fadeUp} transition={{ delay }} className={className}>
   264→      {children}
   265→    </motion.div>
   266→  );
   267→}

   269→// ─── COUNTER ───
   270→function Counter({ target, suffix = "" }: { target: number; suffix?: string }) {
   271→  const [count, setCount] = useState(0);
   272→  const ref = useRef<HTMLSpanElement>(null);
   273→  const isInView = useInView(ref, { once: true });
   274→  useEffect(() => {
   275→    if (!isInView) return;
   276→    let start = 0;
   277→    const step = Math.max(Math.ceil(target / 60), 1);
   278→    const timer = setInterval(() => {
   279→      start += step;
   280→      if (start >= target) { setCount(target); clearInterval(timer); }
   281→      else setCount(start);
   282→    }, 30);
   283→    return () => clearInterval(timer);
   284→  }, [isInView, target]);
   285→  return <span ref={ref}>{count}{suffix}</span>;
   286→}
