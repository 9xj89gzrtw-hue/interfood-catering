   150→      transition={{ delay }}
   151→      className={className}

   197→  /* ─── Validation ─── */
   198→  function validate(): FormErrors {
   199→    const errs: FormErrors = {};
   200→    if (!form.name.trim()) errs.name = "Введите ваше имя";
   201→    if (!form.phone.trim()) {
   202→      errs.phone = "Введите номер телефона";
   203→    } else {
   204→      const phoneClean = form.phone.replace(/[\s\-\(\)]/g, "");
   205→      if (!/^(\+7|7|8)\d{10}$/.test(phoneClean)) {
   206→        errs.phone = "Формат: +7 (XXX) XXX-XX-XX";
   207→      }
   208→    }
   209→    if (form.guests) {
   210→      const n = parseInt(form.guests, 10);
   211→      if (isNaN(n) || n < 1) errs.guests = "Введите количество персон";
   212→    }
   213→    return errs;
   214→  }

   216→  /* ─── Inline validation on blur ─── */
   217→  function validateField(name: keyof FormErrors) {
   218→    const newErrors = { ...errors };

   220→      newErrors.name = "Введите ваше имя";
   221→    } else if (name === "name") {
   222→      delete newErrors.name;
   223→    }
   224→    if (name === "phone") {
   225→      if (!form.phone.trim()) {
   226→        newErrors.phone = "Введите номер телефона";
   227→      } else {
   228→        const phoneClean = form.phone.replace(/[\s\-\(\)]/g, "");
   229→        if (!/^(\+7|7|8)\d{10}$/.test(phoneClean)) {
   230→          newErrors.phone = "Формат: +7 (XXX) XXX-XX-XX";
   231→        } else {
   232→          delete newErrors.phone;
   233→        }
   234→      }
   235→    }
   236→    if (name === "guests" && form.guests) {
   237→      const n = parseInt(form.guests, 10);
   238→      if (isNaN(n) || n < 1) {
   239→        newErrors.guests = "Введите количество персон";

   247→  /* ─── Phone number masking ─── */
   248→  function formatPhone(value: string): string {
   249→    const digits = value.replace(/\D/g, "");
   250→    if (!digits) return "";
   251→    let result = "";
   252→    if (digits.startsWith("8")) {
   253→      result = "+7";
   254→    } else if (digits.startsWith("7")) {
   255→      result = "+7";
   256→    } else if (digits.startsWith("9") || digits.startsWith("3")) {
   257→      result = "+7" + digits[0];
   258→    } else {
   259→      return value; // let them type whatever
   260→    }
   261→
   262→    const rest = digits.startsWith("8") || digits.startsWith("7")
   263→      ? digits.slice(1)
   264→      : digits;
   265→
   266→    if (rest.length > 0) result += " (" + rest.slice(0, 3);
   267→    if (rest.length >= 3) result += ") ";
   268→    if (rest.length > 3) result += rest.slice(3, 6);
   269→    if (rest.length > 6) result += "-" + rest.slice(6, 8);
   270→    if (rest.length > 8) result += "-" + rest.slice(8, 10);
   271→
   272→    return result;
   273→  }

   298→        body: JSON.stringify({
   299→          name: form.name.trim(),
   300→          phone: form.phone.trim(),
   301→          date: form.date || undefined,
   302→          guests: form.guests ? parseInt(form.guests, 10) : undefined,
   303→          venue: form.venue.trim() || undefined,
   304→          source: "website",
   305→        }),
   306→      });
   307→
   308→      const data = await res.json();

   311→        // Show server-side validation error if any
   312→        const serverError = data.error || "Произошла ошибка при отправке. Попробуйте позже.";

   314→        setSubmitting(false);
   315→        return;
   316→      }

   318→      setSubmitted(true);
   319→      setForm(INITIAL_FORM);
   320→      setErrors({});

   324→      setSubmitting(false);
