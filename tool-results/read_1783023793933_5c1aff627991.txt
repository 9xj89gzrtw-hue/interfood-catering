    66→    66→function ParticleCanvas({ isMobile }: { isMobile: boolean }) {
    67→    67→  const canvasRef = useRef<HTMLCanvasElement>(null);
    68→    68→  const animRef = useRef<number>(0);
    69→    69→
    70→    70→  useEffect(() => {
    71→    71→    const canvas = canvasRef.current;
    72→    72→    if (!canvas) return;
    73→    73→    const ctx = canvas.getContext("2d");
    74→    74→    if (!ctx) return;
    75→    75→
    76→    76→    const count = isMobile ? 18 : 35;
    77→    77→    const size = isMobile ? 3 : 2;
    78→    78→    interface Particle {
    79→    79→      x: number;
    80→    80→      y: number;
    81→    81→      r: number;
    82→    82→      vx: number;
    83→    83→      vy: number;
    84→    84→      opacity: number;
    85→    85→      phase: number;
    86→    86→    }
    87→    87→    const particles: Particle[] = [];
    88→    88→
    89→    89→    const resize = () => {
    90→    90→      canvas.width = canvas.offsetWidth * devicePixelRatio;
    91→    91→      canvas.height = canvas.offsetHeight * devicePixelRatio;
    92→    92→      ctx.scale(devicePixelRatio, devicePixelRatio);
    93→    93→    };
    94→    94→    resize();
    95→    95→    window.addEventListener("resize", resize);
    96→    96→
    97→    97→    // Initialize particles
    98→    98→    const w = canvas.offsetWidth;
    99→    99→    const h = canvas.offsetHeight;
   100→   100→    for (let i = 0; i < count; i++) {
   101→   101→      particles.push({
   102→   102→        x: Math.random() * w,
   103→   103→        y: Math.random() * h,
   104→   104→        r: Math.random() * size + 1,
   105→   105→        vx: (Math.random() - 0.5) * 0.3,
   106→   106→        vy: (Math.random() - 0.5) * 0.2 - 0.1,
   107→   107→        opacity: 0.15 + Math.random() * 0.1,
   108→   108→        phase: Math.random() * Math.PI * 2,
   109→   109→      });
   110→   110→    }
   111→   111→
   112→   112→    const draw = (time: number) => {
   113→   113→      ctx.clearRect(0, 0, canvas.offsetWidth, canvas.offsetHeight);
   114→   114→      for (const p of particles) {
   115→   115→        p.x += p.vx;
   116→   116→        p.y += p.vy;
   117→   117→        const pulse = Math.sin(time * 0.001 + p.phase) * 0.05;
   118→   118→        const alpha = Math.max(0.05, p.opacity + pulse);
   119→   119→
   120→   120→        // Wrap around
   121→   121→        if (p.x < -10) p.x = canvas.offsetWidth + 10;
   122→   122→        if (p.x > canvas.offsetWidth + 10) p.x = -10;
   123→   123→        if (p.y < -10) p.y = canvas.offsetHeight + 10;
   124→   124→        if (p.y > canvas.offsetHeight + 10) p.y = -10;
   125→   125→
   126→   126→        ctx.beginPath();
   127→   127→        ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
   128→   128→        ctx.fillStyle = `rgba(212, 166, 62, ${alpha})`;
   129→   129→        ctx.fill();
   130→   130→      }
   131→   131→      animRef.current = requestAnimationFrame(draw);
   132→   132→    };
   133→   133→    animRef.current = requestAnimationFrame(draw);
   134→   134→
   135→   135→    return () => {
   136→   136→      window.removeEventListener("resize", resize);
   137→   137→      cancelAnimationFrame(animRef.current);
   138→   138→    };
   139→   139→  }, [isMobile]);
   140→   140→
   141→   141→  return (
   142→   142→    <canvas
   143→   143→      ref={canvasRef}
   144→   144→      aria-hidden="true"
   145→   145→      style={{
   146→   146→        position: "absolute",
   147→   147→        inset: 0,
   148→   148→        width: "100%",
   149→   149→        height: "100%",
   150→   150→        pointerEvents: "none",
   151→   151→        zIndex: 3,
   152→   152→      }}
   153→   153→    />
   154→   154→  );

   160→   160→function MagneticButton({
   161→   161→  children,
   162→   162→  href,
   163→   163→  variant = "primary",
   164→   164→  isMobile,
   165→   165→}: {
   166→   166→  children: React.ReactNode;
   167→   167→  href: string;
   168→   168→  variant?: "primary" | "secondary";
   169→   169→  isMobile: boolean;
   170→   170→}) {
   171→   171→  const ref = useRef<HTMLAnchorElement>(null);
   172→   172→  const [ripples, setRipples] = useState<{ x: number; y: number; id: number }[]>([]);
   173→   173→  const x = useMotionValue(0);
   174→   174→  const y = useMotionValue(0);
   175→   175→  const springX = useSpring(x, SPRING_MAGNETIC);
   176→   176→  const springY = useSpring(y, SPRING_MAGNETIC);
   177→   177→  const [hovered, setHovered] = useState(false);
   178→   178→
   179→   179→  const handleMouseMove = useCallback(
   180→   180→    (e: React.MouseEvent) => {
   181→   181→      if (isMobile) return;
   182→   182→      const el = ref.current;
   183→   183→      if (!el) return;
   184→   184→      const rect = el.getBoundingClientRect();
   185→   185→      const cx = rect.left + rect.width / 2;
   186→   186→      const cy = rect.top + rect.height / 2;
   187→   187→      const dist = Math.hypot(e.clientX - cx, e.clientY - cy);
   188→   188→      const radius = 120;
   189→   189→      if (dist < radius) {
   190→   190→        const strength = 1 - dist / radius;
   191→   191→        x.set((e.clientX - cx) * strength * 0.3);
   192→   192→        y.set((e.clientY - cy) * strength * 0.3);
   193→   193→      } else {
   194→   194→        x.set(0);
   195→   195→        y.set(0);
   196→   196→      }
   197→   197→    },
   198→   198→    [isMobile, x, y]
   199→   199→  );
   200→   200→
   201→   201→  const handleMouseLeave = useCallback(() => {
   202→   202→    x.set(0);
   203→   203→    y.set(0);
   204→   204→    setHovered(false);
   205→   205→  }, [x, y]);
   206→   206→
   207→   207→  const handleClick = useCallback(
   208→   208→    (e: React.MouseEvent) => {
   209→   209→      const el = ref.current;
   210→   210→      if (!el) return;
   211→   211→      const rect = el.getBoundingClientRect();
   212→   212→      const id = Date.now();
   213→   213→      setRipples((prev) => [...prev, { x: e.clientX - rect.left, y: e.clientY - rect.top, id }]);
   214→   214→      setTimeout(() => setRipples((prev) => prev.filter((r) => r.id !== id)), 600);
   215→   215→    },
   216→   216→    []
   217→   217→  );
   218→   218→
   219→   219→  const isPrimary = variant === "primary";
   220→   220→
   221→   221→  return (
   222→   222→    <motion.a
   223→   223→      ref={ref}
   224→   224→      href={href}
   225→   225→      onMouseMove={handleMouseMove}
   226→   226→      onMouseLeave={handleMouseLeave}
   227→   227→      onMouseEnter={() => setHovered(true)}
   228→   228→      onClick={handleClick}
   229→   229→      style={{
   230→   230→        x: springX,
   231→   231→        y: springY,
   232→   232→        position: "relative",
   233→   233→        overflow: "hidden",
   234→   234→        display: "inline-flex",
   235→   235→        alignItems: "center",
   236→   236→        justifyContent: "center",
   237→   237→        gap: "0.5rem",
   238→   238→        minWidth: isMobile ? "100%" : "auto",
   239→   239→        padding: isMobile ? "1rem 1.5rem" : "1rem 2.25rem",
   240→   240→        minHeight: 48,
   241→   241→        borderRadius: "0.5rem",
   242→   242→        fontFamily: "var(--font-sans)",
   243→   243→        fontWeight: 500,
   244→   244→        fontSize: "0.95rem",
   245→   245→        letterSpacing: "0.02em",
   246→   246→        textDecoration: "none",
   247→   247→        cursor: "pointer",
   248→   248→        border: "none",
   249→   249→        outline: "none",
   250→   250→        background: isPrimary
   251→   251→          ? hovered
   252→   252→            ? "linear-gradient(135deg, #D4A63E, #B8860B)"
   253→   253→            : "linear-gradient(135deg, #B8860B, #9A6F0A)"
   254→   254→          : "transparent",
   255→   255→        color: isPrimary ? "#FFFFFF" : "#B8860B",
   256→   256→        boxShadow: isPrimary
   257→   257→          ? hovered
   258→   258→            ? "0 0 30px rgba(184,134,11,0.35), 0 4px 15px rgba(184,134,11,0.25)"
   259→   259→            : "0 4px 15px rgba(184,134,11,0.2)"
   260→   260→          : hovered
   261→   261→            ? "0 0 20px rgba(184,134,11,0.15)"
   262→   262→            : "none",
   263→   263→        borderWidth: isPrimary ? 0 : 2,
