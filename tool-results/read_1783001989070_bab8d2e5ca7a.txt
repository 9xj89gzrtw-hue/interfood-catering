   205→  /* ─── Validation ─── */
   206→  function validate(): FormErrors {
   207→    const errs: FormErrors = {};
   208→    if (!form.name.trim()) errs.name = "Введите ваше имя";
   209→    if (!form.phone.trim()) {
   210→      errs.phone = "Введите номер телефона";
   211→    } else {
   212→      const phoneClean = form.phone.replace(/[\s\-\(\)]/g, "");
   213→      if (!/^(\+7|7|8)\d{10}$/.test(phoneClean)) {
   214→        errs.phone = "Формат: +7 (XXX) XXX-XX-XX";
   215→      }
   216→    }
   217→    if (form.email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) {
   218→      errs.email = "Введите корректный email";
   219→    }
   220→    if (form.guests) {
   221→      const n = parseInt(form.guests, 10);
   222→      if (isNaN(n) || n < 1) errs.guests = "Введите количество гостей";
   223→    }
   224→    return errs;
   225→  }

   228→  function validateField(name: keyof FormErrors) {
   229→    const newErrors = { ...errors };

   231→      newErrors.name = "Введите ваше имя";
   232→    } else if (name === "name") {
   233→      delete newErrors.name;
   234→    }
   235→    if (name === "phone") {
   236→      if (!form.phone.trim()) {
   237→        newErrors.phone = "Введите номер телефона";
   238→      } else {
   239→        const phoneClean = form.phone.replace(/[\s\-\(\)]/g, "");

   241→          newErrors.phone = "Формат: +7 (XXX) XXX-XX-XX";
   242→        } else {
   243→          delete newErrors.phone;
   244→        }
   245→      }
   246→    }
   247→    if (name === "email" && form.email) {
   248→      if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email)) {
   249→        newErrors.email = "Введите корректный email";

   251→        delete newErrors.email;

   257→        newErrors.guests = "Введите количество гостей";

   260→      }
   261→    }
   262→    setErrors(newErrors);
   263→  }

   265→  /* ─── Phone number masking ─── */
   266→  function formatPhone(value: string): string {
   267→    const digits = value.replace(/\D/g, "");
   268→    if (!digits) return "";
   269→    let result = "";
   270→    if (digits.startsWith("8")) {
   271→      result = "+7";
   272→    } else if (digits.startsWith("7")) {
   273→      result = "+7";
   274→    } else if (digits.startsWith("9") || digits.startsWith("3")) {
   275→      result = "+7" + digits[0];
   276→    } else {
   277→      return value; // let them type whatever
   278→    }
   279→
   280→    const rest = digits.startsWith("8") || digits.startsWith("7")
   281→      ? digits.slice(1)
   282→      : digits;
   283→
   284→    if (rest.length > 0) result += " (" + rest.slice(0, 3);
   285→    if (rest.length >= 3) result += ") ";
   286→    if (rest.length > 3) result += rest.slice(3, 6);
   287→    if (rest.length > 6) result += "-" + rest.slice(6, 8);
   288→    if (rest.length > 8) result += "-" + rest.slice(8, 10);
   289→
   290→    return result;
   291→  }

   309→    if (Object.keys(errs).length > 0) return;

   311→    setSubmitting(true);

   315→        headers: { "Content-Type": "application/json" },
   316→        body: JSON.stringify({
   317→          name: form.name.trim(),
   318→          phone: form.phone.trim(),
   319→          email: form.email.trim() || undefined,
   320→          eventType: form.eventType || undefined,
   321→          date: form.date || undefined,
   322→          guests: form.guests ? parseInt(form.guests, 10) : undefined,
   323→          message: form.message.trim() || undefined,
   324→          source: "website",
   325→        }),
   326→      });

   328→      const data = await res.json();

   331→        // Show server-side validation error if any
   332→        const serverError = data.error || "Произошла ошибка при отправке. Попробуйте позже.";
   333→        setErrors({ name: serverError });
   334→        setSubmitting(false);
   335→        return;
   336→      }

   338→      setSubmitted(true);
   339→      setForm(INITIAL_FORM);
   340→      setErrors({});
   341→    } catch {
   342→      setErrors({ name: "Не удалось подключиться к серверу. Проверьте интернет и попробуйте снова." });
   343→    } finally {
   344→      setSubmitting(false);
   345→    }
