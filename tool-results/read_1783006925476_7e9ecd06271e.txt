   152→/* ─── Animated Price Hook ─── */
   153→
   154→function useAnimatedPrice(target: number) {
   155→  const motionVal = useMotionValue(0);
   156→  const spring = useSpring(motionVal, { stiffness: 120, damping: 30 });
   157→  const display = useTransform(spring, (v: number) => Math.round(v));
   158→  const [shown, setShown] = useState("0");
   159→
   160→  useEffect(() => {
   161→    motionVal.set(target);
   162→  }, [target, motionVal]);
   163→
   164→  useEffect(() => {
   165→    const unsub = display.on("change", (v: number) => {
   166→      setShown(v.toLocaleString("ru-RU"));
   167→    });
   168→    return unsub;
   169→  }, [display]);
   170→
   171→  return shown;
   172→}
   173→
   174→/* ─── Animated Count Hook (for guest count etc.) ─── */
   175→
   176→function useAnimatedCount(target: number) {
   177→  const motionVal = useMotionValue(0);
   178→  const spring = useSpring(motionVal, { stiffness: 300, damping: 35 });
   179→  const display = useTransform(spring, (v: number) => Math.round(v));
   180→  const [shown, setShown] = useState(String(target));
   181→
   182→  useEffect(() => {
   183→    motionVal.set(target);
   184→  }, [target, motionVal]);
   185→
   186→  useEffect(() => {
   187→    const unsub = display.on("change", (v: number) => {
   188→      setShown(String(v));
   189→    });
   190→    return unsub;
   191→  }, [display]);
   192→
   193→  return shown;
   194→}

   196→/* ─── Reveal Component ─── */
   197→
   198→const fadeUp = {
   199→  hidden: { opacity: 0, y: 60 },
   200→  visible: { opacity: 1, y: 0, transition: { duration: 0.8, ease: [0.4, 0, 0.2, 1] as const } },
   201→};
   202→
   203→function Reveal({
   204→  children,
   205→  className = "",
   206→  delay = 0,
   207→}: {
   208→  children: React.ReactNode;
   209→  className?: string;
   210→  delay?: number;
   211→}) {
   212→  const ref = useRef<HTMLDivElement>(null);
   213→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   214→  return (
   215→    <motion.div
   216→      ref={ref}
   217→      initial="hidden"
   218→      animate={isInView ? "visible" : "hidden"}
   219→      variants={fadeUp}
   220→      transition={{ delay }}
   221→      className={className}
   222→    >
   223→      {children}
   224→    </motion.div>
   225→  );
   226→}
