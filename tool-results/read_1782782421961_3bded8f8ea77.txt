   129→   129→function Reveal({ children, className = "", delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   130→   130→  const ref = useRef<HTMLDivElement>(null);
   131→   131→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   132→   132→  return (
   133→   133→    <motion.div
   134→   134→      ref={ref}
   135→   135→      initial="hidden"
   136→   136→      animate={isInView ? "visible" : "hidden"}
   137→   137→      variants={fadeUp}
   138→   138→      transition={{ delay }}
   139→   139→      className={className}
   140→   140→    >
   141→   141→      {children}
   142→   142→    </motion.div>
   143→   143→  );
   144→   144→}
