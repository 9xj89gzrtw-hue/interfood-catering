   195→// ─── Magnetic Submit Button with Ripple ────────────────────
   196→function MagneticSubmitButton({
   197→  onClick,
   198→  disabled = false,
   199→}: {
   200→  onClick: () => void;
   201→  disabled?: boolean;
   202→}) {
   203→  const ref = useRef<HTMLDivElement>(null);
   204→  const [isHovered, setIsHovered] = useState(false);
   205→  const [ripples, setRipples] = useState<
   206→    { x: number; y: number; id: number }[]
   207→  >([]);
   208→
   209→  const x = useMotionValue(0);
   210→  const y = useMotionValue(0);
   211→  const springX = useSpring(x, { stiffness: 200, damping: 20 });
   212→  const springY = useSpring(y, { stiffness: 200, damping: 20 });
   213→
   214→  const handleMouseMove = useCallback(
   215→    (e: React.MouseEvent) => {
   216→      if (!ref.current) return;
   217→      const rect = ref.current.getBoundingClientRect();
   218→      const centerX = rect.left + rect.width / 2;
   219→      const centerY = rect.top + rect.height / 2;
   220→      x.set((e.clientX - centerX) * 0.1);
   221→      y.set((e.clientY - centerY) * 0.1);
   222→    },
   223→    [x, y]
   224→  );
   225→
   226→  const handleMouseLeave = useCallback(() => {
   227→    x.set(0);
   228→    y.set(0);
   229→    setIsHovered(false);
   230→  }, [x, y]);
   231→
   232→  const handleClick = useCallback(
   233→    (e: React.MouseEvent<HTMLButtonElement>) => {
   234→      if (disabled) return;
   235→      const rect = e.currentTarget.getBoundingClientRect();
   236→      const rippleX = e.clientX - rect.left;
   237→      const rippleY = e.clientY - rect.top;
   238→      const id = Date.now();
   239→      setRipples((prev) => [...prev, { x: rippleX, y: rippleY, id }]);
   240→      setTimeout(() => {
   241→        setRipples((prev) => prev.filter((r) => r.id !== id));
   242→      }, 800);
   243→      onClick();
   244→    },
   245→    [onClick, disabled]
   246→  );
   247→
   248→  return (
   249→    <motion.div
   250→      ref={ref}
   251→      onMouseMove={handleMouseMove}
   252→      onMouseEnter={() => setIsHovered(true)}
   253→      onMouseLeave={handleMouseLeave}
   254→      style={{
   255→        x: springX,
   256→        y: springY,
   257→        display: "inline-block",
   258→        width: "100%",
   259→      }}
   260→    >
   261→      <motion.button
   262→        type="submit"
   263→        onClick={handleClick}
   264→        disabled={disabled}
   265→        animate={{
   266→          boxShadow: isHovered
   267→            ? "0 8px 40px rgba(201,169,106,0.3), 0 0 80px rgba(201,169,106,0.15)"
   268→            : "0 4px 20px rgba(201,169,106,0.15)",
   269→        }}
   270→        transition={{ duration: 0.4 }}
   271→        className="btn-gold"
   272→        style={{
   273→          width: "100%",
   274→          minWidth: "44px",
   275→          minHeight: "44px",
   276→          padding: "1.1rem 2rem",
   277→          fontSize: "0.8rem",
   278→          letterSpacing: "0.14em",
   279→          position: "relative",
   280→          overflow: "hidden",
   281→          opacity: disabled ? 0.6 : 1,
   282→          cursor: disabled ? "not-allowed" : "pointer",
   283→        }}
   284→      >
   285→        {disabled ? "Отправка..." : "Отправить заявку"}
   286→        {ripples.map((r) => (
   287→          <span
   288→            key={r.id}
   289→            style={{
   290→              position: "absolute",
   291→              left: r.x,
   292→              top: r.y,
   293→              width: 0,
   294→              height: 0,
   295→              borderRadius: "50%",
   296→              background: "rgba(201, 169, 106, 0.25)",
   297→              transform: "translate(-50%, -50%)",
   298→              animation: "ripple-expand 0.8s ease-out forwards",
   299→              pointerEvents: "none",
   300→            }}
   301→          />
   302→        ))}
   303→      </motion.button>
   304→    </motion.div>
   305→  );
   306→}
