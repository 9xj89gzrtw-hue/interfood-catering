   159→function Reveal({ children, className = "", delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   160→  const ref = useRef<HTMLDivElement>(null);
   161→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   162→  return (
   163→    <motion.div
   164→      ref={ref}
   165→      initial="hidden"
   166→      animate={isInView ? "visible" : "hidden"}
   167→      variants={fadeUp}
   168→      transition={{ delay }}
   169→      className={className}
   170→    >
   171→      {children}
   172→    </motion.div>
   173→  );
   174→}
