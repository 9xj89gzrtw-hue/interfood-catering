    44→gate() {
    45→  local stage="$1" name="$2" result="$3" evidence="${4:-}"
    46→  local critical="${5:-false}"
    47→  TOTAL=$((TOTAL + 1))
    48→  local icon="✅"; local word="PASS"
    49→  if [ "$result" = "FAIL" ]; then
    50→    icon="❌"; word="FAIL"; FAILED=$((FAILED + 1))
    51→    [ "$critical" = "true" ] && CRITICAL=$((CRITICAL + 1))
    52→    STAGES_FAILED="$STAGES_FAILED $stage/$name"
    53→  elif [ "$result" = "SKIP" ]; then
    54→    icon="⏭️"; word="SKIP"; SKIPPED=$((SKIPPED + 1))
    55→  else
    56→    PASSED=$((PASSED + 1))
    57→    STAGES_PASSED="$STAGES_PASSED $stage/$name"
    58→  fi
    59→  echo "  $icon [$stage] $name — $word ${evidence:+│ $evidence}"
    60→  echo "$icon [$stage] $name — $word ${evidence:+│ $evidence}" >> "$REPORT"
    61→  [ -n "$evidence" ] && echo "    → $evidence" >> "$REPORT"
    62→}
