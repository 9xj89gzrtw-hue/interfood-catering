   265→   265→  /* ─── Phone number masking ─── */
   266→   266→  function formatPhone(value: string): string {
   267→   267→    const digits = value.replace(/\D/g, "");
   268→   268→    if (!digits) return "";
   269→   269→    let result = "";
   270→   270→    if (digits.startsWith("8")) {
   271→   271→      result = "+7";
   272→   272→    } else if (digits.startsWith("7")) {
   273→   273→      result = "+7";
   274→   274→    } else if (digits.startsWith("9") || digits.startsWith("3")) {
   275→   275→      result = "+7" + digits[0];
   276→   276→    } else {
   277→   277→      return value; // let them type whatever
   278→   278→    }
   279→   279→
   280→   280→    const rest = digits.startsWith("8") || digits.startsWith("7")
   281→   281→      ? digits.slice(1)
   282→   282→      : digits;
   283→   283→
   284→   284→    if (rest.length > 0) result += " (" + rest.slice(0, 3);
   285→   285→    if (rest.length >= 3) result += ") ";
   286→   286→    if (rest.length > 3) result += rest.slice(3, 6);
   287→   287→    if (rest.length > 6) result += "-" + rest.slice(6, 8);
   288→   288→    if (rest.length > 8) result += "-" + rest.slice(8, 10);
   289→   289→
   290→   290→    return result;
   291→   291→  }
