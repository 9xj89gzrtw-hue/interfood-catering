   249→// ─── REVEAL ───
   250→function Reveal({ children, className = "", delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   251→  const ref = useRef<HTMLDivElement>(null);
   252→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   253→  return (
   254→    <motion.div ref={ref} initial="hidden" animate={isInView ? "visible" : "hidden"} variants={fadeUp} transition={{ delay }} className={className}>
   255→      {children}
   256→    </motion.div>
   257→  );
   258→}

   260→// ─── COUNTER ───
   261→function Counter({ target, suffix = "" }: { target: number; suffix?: string }) {
   262→  const [count, setCount] = useState(0);
   263→  const ref = useRef<HTMLSpanElement>(null);
   264→  const isInView = useInView(ref, { once: true });
   265→  useEffect(() => {
   266→    if (!isInView) return;
   267→    let start = 0;
   268→    const step = Math.max(Math.ceil(target / 60), 1);
   269→    const timer = setInterval(() => {
   270→      start += step;
   271→      if (start >= target) { setCount(target); clearInterval(timer); }
   272→      else setCount(start);
   273→    }, 30);
   274→    return () => clearInterval(timer);
   275→  }, [isInView, target]);
   276→  return <span ref={ref}>{count}{suffix}</span>;
   277→}
