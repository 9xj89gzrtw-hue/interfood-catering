   159→   159→function Reveal({ children, className = "", delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   160→   160→  const ref = useRef<HTMLDivElement>(null);
   161→   161→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   162→   162→  return (
   163→   163→    <motion.div
   164→   164→      ref={ref}
   165→   165→      initial="hidden"
   166→   166→      animate={isInView ? "visible" : "hidden"}
   167→   167→      variants={fadeUp}
   168→   168→      transition={{ delay }}
   169→   169→      className={className}
   170→   170→    >
   171→   171→      {children}
   172→   172→    </motion.div>
   173→   173→  );
   174→   174→}
