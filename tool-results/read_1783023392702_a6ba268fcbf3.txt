   315→/* ═══════════════════════════════════════════════════════════════════════════
   316→   ANIMATED COUNTER — Counts up/down to a target value
   317→   ═══════════════════════════════════════════════════════════════════════════ */
   318→
   319→function AnimatedCounter({ value, prefix = "", suffix = "" }: { value: number; prefix?: string; suffix?: string }) {
   320→  const spring = useSpring(0, { stiffness: 120, damping: 25 });
   321→  const display = useTransform(spring, (v) => {
   322→    const rounded = Math.round(v);
   323→    return prefix + rounded.toLocaleString("ru-RU") + suffix;
   324→  });
   325→
   326→  useEffect(() => {
   327→    spring.set(value);
   328→  }, [value, spring]);
   329→
   330→  return <motion.span>{display}</motion.span>;
   331→}

   353→
   354→function DishCard({
   355→  dish,
   356→  inCart,
   357→  onAdd,
   358→  onRemove,
   359→  onUpdateQty,
   360→  activeCategory,
   361→  isMobile,
   362→  index,
   363→}: {
   364→  dish: DishItem;
   365→  inCart: number;
   366→  onAdd: (dish: DishItem, categoryId: string) => void;
   367→  onRemove: (dishId: string) => void;
   368→  onUpdateQty: (dishId: string, delta: number) => void;
   369→  activeCategory: string;
   370→  isMobile: boolean;
   371→  index: number;
   372→}) {
   373→  const [imgLoaded, setImgLoaded] = useState(false);
   374→  const [isHovered, setIsHovered] = useState(false);
   375→
   376→  // 3D tilt motion values
   377→  const x = useMotionValue(0);
   378→  const y = useMotionValue(0);
   379→  const rotateX = useTransform(y, [-0.5, 0.5], [6, -6]);
   380→  const rotateY = useTransform(x, [-0.5, 0.5], [-6, 6]);
   381→
   382→  /** Handle mouse move for 3D tilt effect */
   383→  const handleMouseMove = useCallback(
   384→    (e: React.MouseEvent<HTMLDivElement>) => {
   385→      if (isMobile) return;
   386→      const rect = e.currentTarget.getBoundingClientRect();
   387→      const px = (e.clientX - rect.left) / rect.width - 0.5;
   388→      const py = (e.clientY - rect.top) / rect.height - 0.5;
   389→      x.set(px);
   390→      y.set(py);
   391→    },
   392→    [isMobile, x, y]
   393→  );
   394→
   395→  const handleMouseLeave = useCallback(() => {
   396→    x.set(0);
   397→    y.set(0);
   398→    setIsHovered(false);
   399→  }, [x, y]);
   400→
   401→  return (
   402→    <motion.div
   403→      initial={{ opacity: 0, y: 30, scale: 0.97 }}
   404→      animate={{ opacity: 1, y: 0, scale: 1 }}
   405→      transition={{
   406→        duration: 0.5,
   407→        delay: index * 0.08,
   408→        ease: [0.25, 0.46, 0.45, 0.94],
   409→      }}
   410→      className="group relative"
   411→      style={{
   412→        perspective: "800px",
   413→      }}
   414→    >
   415→      <motion.div
   416→        style={{
   417→          rotateX: isMobile ? 0 : rotateX,
   418→          rotateY: isMobile ? 0 : rotateY,
   419→          transformStyle: "preserve-3d",
   420→        }}
   421→        onMouseMove={handleMouseMove}
   422→        onMouseEnter={() => setIsHovered(true)}
   423→        onMouseLeave={handleMouseLeave}
   424→        className="relative overflow-hidden"
   425→        /* ─── Card shell ─── */
   426→      >
   427→        {/* Animated gold gradient border on hover */}
   428→        <motion.div
   429→          animate={{
   430→            opacity: isHovered || inCart > 0 ? 1 : 0,
   431→          }}
   432→          transition={{ duration: 0.4 }}
   433→          className="absolute inset-0 rounded-2xl"
   434→          style={{
   435→            padding: "2px",
   436→            background: isHovered
   437→              ? "linear-gradient(135deg, #B8860B, #D4A63E, #E5BF65, #D4A63E, #B8860B)"
   438→              : inCart > 0
   439→                ? "#B8860B"
   440→                : "transparent",
   441→            backgroundSize: isHovered ? "300% 300%" : "100% 100%",
   442→            animation: isHovered ? "goldBorderShift 3s ease infinite" : "none",
   443→          }}
   444→        >
   445→          <div
   446→            className="w-full h-full rounded-2xl"
   447→            style={{ background: COLORS.card }}
   448→          />
   449→        </motion.div>
   450→
   451→        {/* Inner card content */}
   452→        <div
   453→          className="relative rounded-2xl overflow-hidden"
   454→          style={{
   455→            background: COLORS.card,
   456→            boxShadow: isHovered
   457→              ? "0 20px 60px rgba(0,0,0,0.12), 0 8px 24px rgba(184,134,11,0.08)"
   458→              : "0 2px 12px rgba(0,0,0,0.04)",
   459→            transition: "box-shadow 0.4s ease",
   460→          }}
   461→        >
   462→          <div
   463→            className={`flex ${isMobile ? "flex-col" : "flex-col sm:flex-row"}`}

   465→            {/* ─── Image section ─── */}

   469→              }`}
   470→              style={{ transformStyle: "preserve-3d" }}
   471→            >
   472→              {!imgLoaded && <ImageSkeleton />}
   473→              <Image
   474→                src={dish.image}

   477→                className={`object-cover transition-transform duration-700 group-hover:scale-110 ${
