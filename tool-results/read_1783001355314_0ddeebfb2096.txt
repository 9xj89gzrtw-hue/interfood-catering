     9→     9→## 1. View Transitions API in Next.js App Router
    10→    10→
    11→    11→### Current State in This Project
    12→    12→
    13→    13→- **`next.config.ts`** has `experimental.viewTransition: true` ✅
    14→    14→- **`ViewTransitionLink.tsx`** manually calls `document.startViewTransition()` ✅
    15→    15→- **CRITICAL BUG:** The two approaches **CONFLICT** — when Next.js's built-in viewTransition flag is enabled, it wraps `router.push()` in its own `startViewTransition()`. The ViewTransitionLink component then wraps it in a SECOND `startViewTransition()`, creating **nested transitions that break animations**.
    16→    16→
    17→    17→### How Next.js 16's Built-in View Transitions Work
    18→    18→
    19→    19→When `experimental.viewTransition: true` is set, Next.js automatically wraps all App Router client-side navigations in `document.startViewTransition()`. This means:
    20→    20→
    21→    21→1. You do NOT need a custom ViewTransitionLink — regular `<Link>` gets transitions automatically
    22→    22→2. The old/outgoing snapshot is `::view-transition-old(root)`, the new/incoming is `::view-transition-new(root)`
    23→    23→3. You style the transition purely with CSS
    24→    24→
    25→    25→### ✅ Recommended Implementation (Fix the Current Setup)
    26→    26→
    27→    27→#### Step 1: Remove the manual ViewTransitionLink OR disable the experimental flag
    28→    28→
    29→    29→**Option A (RECOMMENDED): Use Next.js built-in + CSS-only animations**
    30→    30→
    31→    31→```css
    32→    32→/* globals.css — Pure CSS View Transitions with Next.js 16 */
    33→    33→
    34→    34→/* Disable the default cross-fade */
    35→    35→::view-transition-old(root),
    36→    36→::view-transition-new(root) {
    37→    37→  animation: none;
    38→    38→  mix-blend-mode: normal;
    39→    39→}
    40→    40→
    41→    41→/* Custom page transition: slide + fade */
    42→    42→::view-transition-old(root) {
    43→    43→  animation: 0.4s ease-in both vt-slide-out;
    44→    44→}
    45→    45→::view-transition-new(root) {
    46→    46→  animation: 0.4s ease-out both vt-slide-in;
    47→    47→}
    48→    48→
    49→    49→@keyframes vt-slide-out {
    50→    50→  to {
    51→    51→    opacity: 0;
    52→    52→    transform: translateX(-30px) scale(0.98);
    53→    53→  }
    54→    54→}
    55→    55→@keyframes vt-slide-in {
    56→    56→  from {
    57→    57→    opacity: 0;
    58→    58→    transform: translateX(30px) scale(0.98);
    59→    59→  }
    60→    60→}
    61→    61→
    62→    62→/* Circle-clip reveal from click position (the "secret hack") */
    63→    63→/* This requires a tiny bit of JS to set --vt-x and --vt-y on navigation */
    64→    64→::view-transition-new(root) {
    65→    65→  animation: 0.5s cubic-bezier(0.25, 1, 0.5, 1) both vt-circle-reveal;
    66→    66→}
    67→    67→
    68→    68→@keyframes vt-circle-reveal {
    69→    69→  from {
    70→    70→    clip-path: circle(0% at var(--vt-x, 50%) var(--vt-y, 50%));
    71→    71→    opacity: 0.8;
    72→    72→  }
    73→    73→  to {
    74→    74→    clip-path: circle(150% at var(--vt-x, 50%) var(--vt-y, 50%));
    75→    75→    opacity: 1;
    76→    76→  }
    77→    77→}
    78→    78→```

    80→    80→#### Step 2: Minimal JS for click-position tracking
    81→    81→
    82→    82→Since Next.js 16 handles the `startViewTransition()` call internally, we only need to set the CSS custom properties for origin-based transitions. Replace the entire `ViewTransitionLink.tsx` with:
    83→    83→
    84→    84→```tsx
    85→    85→// src/components/ViewTransitionOrigin.tsx — Minimal click-position tracker
    86→    86→"use client";
    87→    87→
    88→    88→import { useCallback, useEffect } from "react";
    89→    89→
    90→    90→/**
    91→    91→ * Tracks click positions and sets --vt-x, --vt-y CSS custom properties
    92→    92→ * on the <html> element so CSS View Transition keyframes can use them.
    93→    93→ * Works WITH Next.js 16's built-in experimental.viewTransition.
    94→    94→ */
    95→    95→export default function ViewTransitionOrigin() {
    96→    96→  const handleMouseMove = useCallback((e: MouseEvent) => {
    97→    97→    document.documentElement.style.setProperty("--vt-x", `${e.clientX}px`);
    98→    98→    document.documentElement.style.setProperty("--vt-y", `${e.clientY}px`);
    99→    99→  }, []);
   100→   100→
   101→   101→  useEffect(() => {
   102→   102→    document.addEventListener("mousemove", handleMouseMove, { passive: true });
   103→   103→    return () => document.removeEventListener("mousemove", handleMouseMove);
   104→   104→  }, [handleMouseMove]);
   105→   105→
   106→   106→  return null; // Renderless component
   107→   107→}
   108→   108→```
   109→   109→
   110→   110→Add to `layout.tsx`:
   111→   111→```tsx
   112→   112→<ViewTransitionOrigin />
   113→   113→```

   421→   421→### Haptic Feedback — Advanced Patterns
   422→   422→
   423→   423→```tsx
   424→   424→// src/hooks/use-haptic.ts — Production haptic hook
   425→   425→"use client";
   426→   426→
   427→   427→type HapticStyle = "light" | "medium" | "heavy" | "selection" | "success" | "warning" | "error";
   428→   428→
   429→   429→export function useHaptic() {
   430→   430→  const vibrate = (style: HapticStyle) => {
   431→   431→    // navigator.vibrate is Android-only
   432→   432→    if (typeof navigator !== "undefined" && "vibrate" in navigator) {
   433→   433→      const patterns: Record<HapticStyle, number | number[]> = {
   434→   434→        light: [10],
   435→   435→        medium: [20],
   436→   436→        heavy: [30, 10, 30],
   437→   437→        selection: [5],
   438→   438→        success: [10, 50, 20],
   439→   439→        warning: [20, 50, 20, 50, 20],
   440→   440→        error: [30, 50, 30, 50, 30, 50, 30],
   441→   441→      };
   442→   442→      navigator.vibrate(patterns[style]);
   443→   443→    }
   444→   444→
   445→   445→    // iOS: Use the Vibration API if available (iOS 17+)
   446→   446→    if (typeof navigator !== "undefined" && "vibrate" in navigator === false) {
   447→   447→      // Fallback: visual feedback for iOS
   448→   448→      // iOS Safari does NOT support navigator.vibrate
   449→   449→      // The only option is visual: flash/scale the element
   450→   450→    }
   451→   451→  };
   452→   452→
   453→   453→  return { vibrate };
   454→   454→}
   455→   455→
   456→   456→// Usage in components:
   457→   457→// const { vibrate } = useHaptic();
   458→   458→// <button onClick={() => vibrate("light")}>Tap me</button>

   486→   486→import { useEffect, useRef } from "react";
   487→   487→
   488→   488→export function useSwipeBack(threshold = 100) {
   489→   489→  const router = useRouter();
   490→   490→  const startX = useRef(0);
   491→   491→  const startY = useRef(0);
   492→   492→
   493→   493→  useEffect(() => {
   494→   494→    // Only on touch devices
   495→   495→    if (!("ontouchstart" in window)) return;
   496→   496→
   497→   497→    const handleStart = (e: TouchEvent) => {
   498→   498→      // Only trigger from left edge (like iOS)
   499→   499→      if (e.touches[0].clientX < 30) {
   500→   500→        startX.current = e.touches[0].clientX;
   501→   501→        startY.current = e.touches[0].clientY;
   502→   502→      }
   503→   503→    };
   504→   504→
   505→   505→    const handleEnd = (e: TouchEvent) => {
   506→   506→      const dx = e.changedTouches[0].clientX - startX.current;
   507→   507→      const dy = e.changedTouches[0].clientY - startY.current;
   508→   508→      // Must be mostly horizontal, from left edge, with sufficient distance
   509→   509→      if (dx > threshold && Math.abs(dx) > Math.abs(dy) * 2) {
   510→   510→        router.back();
   511→   511→      }
   512→   512→    };
   513→   513→
   514→   514→    document.addEventListener("touchstart", handleStart, { passive: true });
   515→   515→    document.addEventListener("touchend", handleEnd, { passive: true });
   516→   516→
   517→   517→    return () => {
   518→   518→      document.removeEventListener("touchstart", handleStart);
   519→   519→      document.removeEventListener("touchend", handleEnd);
   520→   520→    };
   521→   521→  }, [router, threshold]);
   522→   522→}
   523→   523→```
   524→   524→
   525→   525→### Touch-Enhanced UI Patterns
   526→   526→
   527→   527→```tsx
   528→   528→// 1. Pull-to-refresh pattern
   529→   529→const pullDistance = useRef(0);
   530→   530→const PULL_THRESHOLD = 80;
   531→   531→
   532→   532→onTouchStart → record startY
   533→   533→onTouchMove → {

   726→   726→### 🔥 Hack 3: Lenis + Scroll-Driven Animations Integration
   727→   727→
   728→   728→The issue: Lenis virtualizes scroll position, which breaks `animation-timeline: scroll()`. The fix:
   729→   729→
   730→   730→```css
   731→   731→/* Lenis uses transform on the wrapper, which breaks scroll() timeline */
   732→   732→/* Solution: Use view() timeline instead, which works with Lenis! */
   733→   733→/* Or: configure Lenis to use native scroll */
   734→   734→```
   735→   735→
   736→   736→```tsx
   737→   737→// In SmoothScroll.tsx, add:
   738→   738→const lenis = new Lenis({
   739→   739→  // ... existing config ...
   740→   740→  // CRITICAL: This makes Lenis use native scrolling instead of transforms
   741→   741→  // which allows animation-timeline: scroll() to work
   742→   742→  __experimentalScrollMode: 'native', // Lenis 1.3+ 
   743→   743→});
   744→   744→```
   745→   745→
   746→   746→Actually, the real fix is simpler — use `animation-timeline: view()` (which works regardless of scroll method) instead of `scroll()` where possible. The `view()` timeline is based on element visibility, not scroll position.
