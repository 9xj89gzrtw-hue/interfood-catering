   193→// ─── Magnetic Submit Button with Ripple ────────────────────
   194→function MagneticSubmitButton({
   195→  onClick,
   196→  disabled = false,
   197→}: {
   198→  onClick: () => void;
   199→  disabled?: boolean;
   200→}) {
   201→  const ref = useRef<HTMLDivElement>(null);
   202→  const [isHovered, setIsHovered] = useState(false);
   203→  const [ripples, setRipples] = useState<
   204→    { x: number; y: number; id: number }[]
   205→  >([]);
   206→
   207→  const x = useMotionValue(0);
   208→  const y = useMotionValue(0);
   209→  const springX = useSpring(x, { stiffness: 200, damping: 20 });
   210→  const springY = useSpring(y, { stiffness: 200, damping: 20 });
   211→
   212→  const handleMouseMove = useCallback(
   213→    (e: React.MouseEvent) => {
   214→      if (!ref.current) return;
   215→      const rect = ref.current.getBoundingClientRect();
   216→      const centerX = rect.left + rect.width / 2;
   217→      const centerY = rect.top + rect.height / 2;
   218→      x.set((e.clientX - centerX) * 0.1);
   219→      y.set((e.clientY - centerY) * 0.1);
   220→    },
   221→    [x, y]
   222→  );
   223→
   224→  const handleMouseLeave = useCallback(() => {
   225→    x.set(0);
   226→    y.set(0);
   227→    setIsHovered(false);
   228→  }, [x, y]);
   229→
   230→  const handleClick = useCallback(
   231→    (e: React.MouseEvent<HTMLButtonElement>) => {
   232→      if (disabled) return;
   233→      const rect = e.currentTarget.getBoundingClientRect();
   234→      const rippleX = e.clientX - rect.left;
   235→      const rippleY = e.clientY - rect.top;
   236→      const id = Date.now();
   237→      setRipples((prev) => [...prev, { x: rippleX, y: rippleY, id }]);
   238→      setTimeout(() => {
   239→        setRipples((prev) => prev.filter((r) => r.id !== id));
   240→      }, 800);
   241→      onClick();
   242→    },
   243→    [onClick, disabled]
   244→  );
   245→
   246→  return (
   247→    <motion.div
   248→      ref={ref}
   249→      onMouseMove={handleMouseMove}
   250→      onMouseEnter={() => setIsHovered(true)}
   251→      onMouseLeave={handleMouseLeave}
   252→      style={{
   253→        x: springX,
   254→        y: springY,
   255→        display: "inline-block",
   256→        width: "100%",
   257→      }}
   258→    >
   259→      <motion.button
   260→        type="submit"
   261→        onClick={handleClick}
   262→        disabled={disabled}
   263→        animate={{
   264→          boxShadow: isHovered
   265→            ? "0 8px 40px rgba(201,169,106,0.3), 0 0 80px rgba(201,169,106,0.15)"
   266→            : "0 4px 20px rgba(201,169,106,0.15)",
   267→        }}
   268→        transition={{ duration: 0.4 }}
   269→        className="btn-gold"
   270→        style={{
   271→          width: "100%",
   272→          minWidth: "44px",
   273→          minHeight: "44px",
   274→          padding: "1.1rem 2rem",
   275→          fontSize: "0.8rem",
   276→          letterSpacing: "0.14em",
   277→          position: "relative",
   278→          overflow: "hidden",
   279→          opacity: disabled ? 0.6 : 1,
   280→          cursor: disabled ? "not-allowed" : "pointer",
   281→        }}
   282→      >
   283→        {disabled ? "Отправка..." : "Отправить заявку"}
   284→        {ripples.map((r) => (
   285→          <span
   286→            key={r.id}
   287→            style={{
   288→              position: "absolute",
   289→              left: r.x,
   290→              top: r.y,
   291→              width: 0,
   292→              height: 0,
   293→              borderRadius: "50%",
   294→              background: "rgba(201, 169, 106, 0.25)",
   295→              transform: "translate(-50%, -50%)",
   296→              animation: "ripple-expand 0.8s ease-out forwards",
   297→              pointerEvents: "none",
   298→            }}
   299→          />
   300→        ))}
   301→      </motion.button>
   302→    </motion.div>
   303→  );
   304→}
