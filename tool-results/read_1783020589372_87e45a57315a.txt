   356→/* ═══════════════════════════════════════════════════════════════

   359→
   360→export default function MenuBuilder() {
   361→  const isMobile = useIsMobile();
   362→  const [activeCategory, setActiveCategory] = useState("furshet");
   363→  const [cart, setCart] = useState<CartItem[]>([]);
   364→  const [guests, setGuests] = useState(50);
   365→  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
   366→  const [pdfGenerating, setPdfGenerating] = useState(false);
   367→  const categoryScrollRef = useRef<HTMLDivElement>(null);
   368→
   369→  /* ─── Cart operations ─── */
   370→  const addToCart = useCallback((dish: DishItem, categoryId: string) => {
   371→    setCart((prev) => {
   372→      const existing = prev.find((item) => item.dish.id === dish.id);
   373→      if (existing) {
   374→        return prev.map((item) =>
   375→          item.dish.id === dish.id
   376→            ? { ...item, quantity: item.quantity + 1 }
   377→            : item
   378→        );
   379→      }
   380→      return [...prev, { dish, quantity: 1, categoryId }];
   381→    });
   382→  }, []);
   383→
   384→  const removeFromCart = useCallback((dishId: string) => {
   385→    setCart((prev) => prev.filter((item) => item.dish.id !== dishId));
   386→  }, []);
   387→
   388→  const updateQuantity = useCallback((dishId: string, delta: number) => {
   389→    setCart((prev) =>
   390→      prev
   391→        .map((item) =>
   392→          item.dish.id === dishId
   393→            ? { ...item, quantity: Math.max(0, item.quantity + delta) }
   394→            : item
   395→        )
   396→        .filter((item) => item.quantity > 0)
   397→    );
   398→  }, []);
   399→
   400→  const getCartQuantity = useCallback(
   401→    (dishId: string) => {
   402→      return cart.find((item) => item.dish.id === dishId)?.quantity || 0;
   403→    },
   404→    [cart]
   405→  );
   406→
   407→  const totalItems = useMemo(
   408→    () => cart.reduce((sum, item) => sum + item.quantity, 0),
   409→    [cart]
   410→  );
   411→
   412→  const costPerPerson = useMemo(
   413→    () => cart.reduce((sum, item) => sum + item.dish.pricePerPerson * item.quantity, 0),
   414→    [cart]
   415→  );
   416→
   417→  const discountPercent = useMemo(() => {
   418→    if (guests >= 200) return 15;
   419→    if (guests >= 100) return 10;
   420→    return 0;
   421→  }, [guests]);
   422→
   423→  const discountAmount = useMemo(
   424→    () => Math.round(costPerPerson * guests * (discountPercent / 100)),
   425→    [costPerPerson, guests, discountPercent]
   426→  );
   427→
   428→  const totalCost = useMemo(
   429→    () => costPerPerson * guests - discountAmount,
   430→    [costPerPerson, guests, discountAmount]
   431→  );
   432→
   433→  /* ─── PDF generation with jspdf ─── */
   434→  const generatePDF = useCallback(async () => {
   435→    setPdfGenerating(true);
   436→    try {
   437→      const { jsPDF } = await import("jspdf");
   438→      const doc = new jsPDF({ orientation: "portrait", unit: "mm", format: "a4" });
   439→
   440→      const pageWidth = doc.internal.pageSize.getWidth();
   441→      const margin = 20;
   442→      const contentWidth = pageWidth - margin * 2;
   443→      let y = margin;
   444→
   445→      // Helper: check page overflow
   446→      const checkPage = (needed: number) => {
   447→        if (y + needed > doc.internal.pageSize.getHeight() - 20) {
   448→          doc.addPage();
   449→          y = margin;
   450→        }
   451→      };
   452→
   453→      // ─── Company Header ───
   454→      doc.setFont("helvetica", "bold");
   455→      doc.setFontSize(22);
   456→      doc.setTextColor(184, 134, 11); // #B8860B
   457→      doc.text("INTERFOOD CATERING", pageWidth / 2, y, { align: "center" });
   458→      y += 8;
   459→
   460→      doc.setFont("helvetica", "normal");
   461→      doc.setFontSize(11);
   462→      doc.setTextColor(125, 90, 13); // #7D5A0D
   463→      doc.text("Ваше персональное меню", pageWidth / 2, y, { align: "center" });
   464→      y += 4;
   465→
   466→      // Gold divider line
   467→      doc.setDrawColor(184, 134, 11);
   468→      doc.setLineWidth(0.8);
   469→      doc.line(margin, y, pageWidth - margin, y);
   470→      y += 10;
   471→
   472→      // ─── Menu items by category ───
   473→      const categoryMap: Record<string, string> = {};
   474→      MENU_CATEGORIES.forEach((cat) => {
   475→        cat.items.forEach((item) => {
   476→          categoryMap[item.id] = cat.name;
   477→        });
   478→      });
   479→
   480→      const groupedByCategory: Record<string, CartItem[]> = {};
