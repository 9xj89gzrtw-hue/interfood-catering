   249→export default function QuizPage() {
   250→  const [currentStep, setCurrentStep] = useState(0);
   251→  const [answers, setAnswers] = useState<Record<number, string>>({});
   252→  const [direction, setDirection] = useState(1);
   253→  const [showResults, setShowResults] = useState(false);
   254→  const [focusedIndex, setFocusedIndex] = useState(0);
   255→  const [contactForm, setContactForm] = useState({ name: "", phone: "" });
   256→  const [formSubmitted, setFormSubmitted] = useState(false);
   257→  const [isSubmitting, setIsSubmitting] = useState(false);
   258→  const [toast, setToast] = useState<string | null>(null);
   259→
   260→  const quizContainerRef = useRef<HTMLDivElement>(null);
   261→  const optionsRef = useRef<(HTMLButtonElement | null)[]>([]);
   262→
   263→  const totalSteps = STEPS.length;
   264→  const isLastStep = currentStep === totalSteps - 1;
   265→  const hasAnswer = answers[currentStep] !== undefined;
   266→
   267→  /* Toast auto-dismiss */
   268→  useEffect(() => {
   269→    if (!toast) return;
   270→    const t = setTimeout(() => setToast(null), 4000);
   271→    return () => clearTimeout(t);
   272→  }, [toast]);
   273→
   274→  const selectOption = useCallback(
   275→    (value: string) => {
   276→      setAnswers((prev) => ({ ...prev, [currentStep]: value }));
   277→    },
   278→    [currentStep]
   279→  );
   280→
   281→  /* Keyboard navigation */
   282→  useEffect(() => {
   283→    if (showResults) return;
   284→
   285→    const handleKeyDown = (e: KeyboardEvent) => {
   286→      const step = STEPS[currentStep];
   287→      if (!step) return;
   288→      const maxIdx = step.options.length - 1;
   289→
   290→      if (e.key === "ArrowDown" || e.key === "ArrowRight") {
   291→        e.preventDefault();
   292→        setFocusedIndex((prev) => (prev < maxIdx ? prev + 1 : 0));
   293→      } else if (e.key === "ArrowUp" || e.key === "ArrowLeft") {
   294→        e.preventDefault();
   295→        setFocusedIndex((prev) => (prev > 0 ? prev - 1 : maxIdx));
   296→      } else if (e.key === "Enter" || e.key === " ") {
   297→        e.preventDefault();
   298→        selectOption(step.options[focusedIndex].value);
   299→      } else if (e.key === "Tab") {
   300→        // Allow Tab to leave quiz options so users can reach nav/action buttons
   301→      }
   302→    };
   303→
   304→    window.addEventListener("keydown", handleKeyDown);
   305→    return () => window.removeEventListener("keydown", handleKeyDown);
   306→  }, [currentStep, focusedIndex, showResults, selectOption]);
   307→
   308→  /* Focus the option button when focusedIndex changes */
   309→  useEffect(() => {
   310→    const el = optionsRef.current[focusedIndex];
   311→    if (el && !showResults) {
   312→      el.focus();
   313→    }
   314→  }, [focusedIndex, currentStep, showResults]);
   315→
   316→  /* Reset focusedIndex when step changes */
   317→  useEffect(() => {
   318→    setFocusedIndex(0);
   319→    optionsRef.current = [];
   320→  }, [currentStep]);
   321→
   322→  const goNext = useCallback(() => {
   323→    if (!hasAnswer) return;
   324→    if (isLastStep) {
   325→      setShowResults(true);
   326→    } else {
   327→      setDirection(1);
   328→      setCurrentStep((prev) => prev + 1);
   329→    }
   330→  }, [hasAnswer, isLastStep]);
   331→
   332→  const goBack = useCallback(() => {
   333→    if (showResults) {
   334→      setShowResults(false);
   335→      return;
   336→    }
   337→    if (currentStep > 0) {
   338→      setDirection(-1);
   339→      setCurrentStep((prev) => prev - 1);
   340→    }
   341→  }, [currentStep, showResults]);
   342→
   343→  /* Auto-advance on select */
   344→  useEffect(() => {
   345→    if (!hasAnswer) return;
   346→    const timer = setTimeout(goNext, 450);
   347→    return () => clearTimeout(timer);
   348→  }, [hasAnswer, goNext]);
   349→
   350→  /* Contact form submit */
   351→  const handleSubmitContact = async () => {
   352→    if (!contactForm.name.trim() || !contactForm.phone.trim()) {
   353→      setToast("Пожалуйста, заполните все поля");
   354→      return;
   355→    }
   356→    setIsSubmitting(true);
   357→    try {
   358→      const res = await fetch("/api/contact", {
   359→        method: "POST",
   360→        headers: { "Content-Type": "application/json" },
   361→        body: JSON.stringify({
   362→          name: contactForm.name,
   363→          phone: contactForm.phone,
   364→          message: `Квиз: рекомендация — ${getRecommendation(answers).title}, бюджет — ${getRecommendation(answers).priceRange}`,
   365→        }),
   366→      });
   367→      if (res.ok) {
   368→        setFormSubmitted(true);
   369→        setToast("Заявка отправлена! Мы свяжемся с вами в ближайшее время.");
   370→      } else {
   371→        setToast("Ошибка отправки. Попробуйте ещё раз.");
   372→      }
   373→    } catch {
   374→      setToast("Ошибка сети. Попробуйте ещё раз.");
   375→    } finally {
   376→      setIsSubmitting(false);
   377→    }
   378→  };
   379→
   380→  const recommendation = getRecommendation(answers);
   381→
   382→  /* ─── Render ─── */
   383→  return (

   391→          display: "flex",
   392→          flexDirection: "column",
   393→        }}
   394→      >
   395→        {/* ── Hero header ── */}
   396→        <section
   397→          style={{
   398→            paddingTop: "clamp(6rem, 12vh, 8rem)",
   399→            paddingBottom: "2rem",
   400→            textAlign: "center",
   401→            background: "var(--color-cream)",
   402→            position: "relative",
   403→            overflow: "hidden",
   404→          }}
