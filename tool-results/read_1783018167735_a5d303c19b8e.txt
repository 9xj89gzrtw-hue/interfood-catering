   141→function Reveal({ children, className = "", delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   142→  const ref = useRef<HTMLDivElement>(null);
   143→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   144→  return (
   145→    <motion.div ref={ref} initial="hidden" animate={isInView ? "visible" : "hidden"} variants={fadeUp} transition={{ delay }} className={className}>
   146→      {children}
   147→    </motion.div>
   148→  );
   149→}
