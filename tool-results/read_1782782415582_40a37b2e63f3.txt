    90→// ─── Reveal helper ───
    91→function Reveal({ children, delay = 0, className = "" }: { children: React.ReactNode; delay?: number; className?: string }) {
    92→  const ref = useRef<HTMLDivElement>(null);
    93→  const inView = useInView(ref, { once: true, margin: "-60px" });
    94→  return (
    95→    <motion.div
    96→      ref={ref}
    97→      className={className}
    98→      initial={{ opacity: 0, y: 40 }}
    99→      animate={inView ? { opacity: 1, y: 0 } : {}}
   100→      transition={{ duration: 0.8, delay, ease: [0.25, 1, 0.5, 1] }}
   101→    >
   102→      {children}
   103→    </motion.div>
   104→  );
   105→}

   108→function VideoCard({ video, index }: { video: typeof VIDEOS[0]; index: number }) {
   109→  const videoRef = useRef<HTMLVideoElement>(null);
   110→  const [isPlaying, setIsPlaying] = useState(false);
   111→
   112→  const handleMouseEnter = () => {
   113→    if (videoRef.current) {
   114→      videoRef.current.play().catch(() => {});
   115→      setIsPlaying(true);
   116→    }
   117→  };
   118→
   119→  const handleMouseLeave = () => {
   120→    if (videoRef.current) {
   121→      videoRef.current.pause();
   122→      videoRef.current.currentTime = 0;
   123→      setIsPlaying(false);
   124→    }
   125→  };

   128→    <Reveal delay={index * 0.1}>
