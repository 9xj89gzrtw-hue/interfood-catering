   161→const fadeUp = {

   166→function Reveal({ children, className = "", delay = 0 }: { children: React.ReactNode; className?: string; delay?: number }) {
   167→  const ref = useRef<HTMLDivElement>(null);
   168→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   169→  return (
   170→    <motion.div
   171→      ref={ref}
   172→      initial="hidden"
   173→      animate={isInView ? "visible" : "hidden"}
   174→      variants={fadeUp}
   175→      transition={{ delay }}
   176→      className={className}
   177→    >
   178→      {children}
   179→    </motion.div>
   180→  );
   181→}
