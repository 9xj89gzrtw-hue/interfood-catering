   124→// ─── Reveal wrapper ───
   125→function Reveal({ children, delay = 0, className = "" }: { children: React.ReactNode; delay?: number; className?: string }) {
   126→  const ref = useRef<HTMLDivElement>(null);
   127→  const inView = useInView(ref, { once: true, margin: "-60px" });
   128→  return (
   129→    <motion.div
   130→      ref={ref}
   131→      className={className}
   132→      initial={{ opacity: 0, y: 50 }}
   133→      animate={inView ? { opacity: 1, y: 0 } : {}}
   134→      transition={{ duration: 0.8, delay, ease: [0.25, 1, 0.5, 1] }}
   135→    >
   136→      {children}
   137→    </motion.div>
   138→  );
   139→}
