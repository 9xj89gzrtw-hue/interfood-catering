   179→   179→function Reveal({
   180→   180→  children,
   181→   181→  className = "",
   182→   182→  delay = 0,
   183→   183→}: {
   184→   184→  children: React.ReactNode;
   185→   185→  className?: string;
   186→   186→  delay?: number;
   187→   187→}) {
   188→   188→  const ref = useRef<HTMLDivElement>(null);
   189→   189→  const isInView = useInView(ref, { once: true, margin: "-60px" });
   190→   190→  return (
   191→   191→    <motion.div
   192→   192→      ref={ref}
   193→   193→      initial="hidden"
   194→   194→      animate={isInView ? "visible" : "hidden"}
   195→   195→      variants={fadeUp}
   196→   196→      transition={{ delay }}
   197→   197→      className={className}
   198→   198→    >
   199→   199→      {children}
   200→   200→    </motion.div>
   201→   201→  );
   202→   202→}
