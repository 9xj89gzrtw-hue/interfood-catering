    72→    72→function MorphingText({ words, interval = 2800 }: { words: string[]; interval?: number }) {
    73→    73→  const [index, setIndex] = useState(0);
    74→    74→  const [visible, setVisible] = useState(true);
    75→    75→
    76→    76→  useEffect(() => {
    77→    77→    const timer = setInterval(() => {
    78→    78→      setVisible(false);
    79→    79→      setTimeout(() => {
    80→    80→        setIndex((prev) => (prev + 1) % words.length);
    81→    81→        setVisible(true);
    82→    82→      }, 450);
    83→    83→    }, interval);
    84→    84→    return () => clearInterval(timer);
    85→    85→  }, [words.length, interval]);
    86→    86→
    87→    87→  return (
    88→    88→    <AnimatePresence mode="wait">
    89→    89→      <motion.span
    90→    90→        key={words[index]}
    91→    91→        initial={{ opacity: 0, filter: "blur(12px)", y: 8, scale: 0.92 }}
    92→    92→        animate={
    93→    93→          visible
    94→    94→            ? { opacity: 1, filter: "blur(0px)", y: 0, scale: 1 }
    95→    95→            : { opacity: 0, filter: "blur(12px)", y: -8, scale: 1.08 }
    96→    96→        }
    97→    97→        exit={{ opacity: 0, filter: "blur(12px)", y: -8, scale: 1.08 }}
    98→    98→        transition={{ duration: 0.5, ease: EASE_PREMIUM }}
    99→    99→        style={{
   100→   100→          display: "inline-block",
   101→   101→          fontStyle: "italic",
   102→   102→          color: "#D4A63E",
   103→   103→          // Fixed min-width based on longest word to prevent layout shift
   104→   104→          minWidth: `${LONGEST_TAGLINE.length + 0.5}ch`,
   105→   105→          textAlign: "center" as const,
   106→   106→          willChange: "opacity, filter, transform",
   107→   107→          textShadow:
   108→   108→            "0 2px 12px rgba(0,0,0,0.6), 0 0 40px rgba(184,134,11,0.3), 0 0 80px rgba(184,134,11,0.15)",
   109→   109→        }}
   110→   110→      >
   111→   111→        {words[index]}
   112→   112→      </motion.span>
   113→   113→    </AnimatePresence>
   114→   114→  );
   115→   115→}

   675→   675→function MouseGlow() {
   676→   676→  const glowX = useMotionValue(0);
   677→   677→  const glowY = useMotionValue(0);
   678→   678→  const springX = useSpring(glowX, { stiffness: 80, damping: 30 });
   679→   679→  const springY = useSpring(glowY, { stiffness: 80, damping: 30 });
   680→   680→  const [isActive, setIsActive] = useState(false);
   681→   681→
   682→   682→  const glowBackground = useTransform(
   683→   683→    [springX, springY],
   684→   684→    ([x, y]: number[]) =>
   685→   685→      `radial-gradient(600px circle at ${x}px ${y}px, rgba(184,134,11,0.08), transparent 60%)`
   686→   686→  );
   687→   687→
   688→   688→  useEffect(() => {
   689→   689→    if (typeof window !== "undefined" && "ontouchstart" in window) return;
   690→   690→    const handleMouseMove = (e: MouseEvent) => {
   691→   691→      glowX.set(e.clientX);
   692→   692→      glowY.set(e.clientY);
   693→   693→      setIsActive(true);
   694→   694→    };
   695→   695→    const handleMouseLeave = () => setIsActive(false);
   696→   696→    window.addEventListener("mousemove", handleMouseMove);
   697→   697→    document.addEventListener("mouseleave", handleMouseLeave);
   698→   698→    return () => {
   699→   699→      window.removeEventListener("mousemove", handleMouseMove);
   700→   700→      document.removeEventListener("mouseleave", handleMouseLeave);
   701→   701→    };
   702→   702→  }, [glowX, glowY]);
   703→   703→
   704→   704→  if (!isActive) return null;
   705→   705→
   706→   706→  return (
   707→   707→    <motion.div
   708→   708→      style={{
   709→   709→        position: "absolute",
   710→   710→        inset: 0,
   711→   711→        pointerEvents: "none",
   712→   712→        zIndex: 2,
   713→   713→        background: glowBackground,
   714→   714→        transition: "opacity 0.5s",
   715→   715→      }}
   716→   716→    />
   717→   717→  );
   718→   718→}
